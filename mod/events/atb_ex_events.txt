namespace = atb_ex

# 全局开局（on_game_start，排在原版 ancrel.12050 随机生成天选民之后）
event = {
	id = atb_ex.1
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		atb_ex_apply_galaxy_rules = yes
		atb_ex_grant_habitat_techs = yes
		# 3.10：银河里至少 7 座 L 星门（开局设置关掉 L 星门时也一样），在远征军星团生成之前做
		atb_ex_ensure_lgates = yes
		# 3.1：第二天再补一次银河规则（开局当天设置的全局旗标、新国家都能接上）
		random_country = { country_event = { id = atb_ex.5 days = 1 } }
		# 3.6：两个虫洞、白鹿与远星的好感（等两个星团都生成好，第三天做）
		random_country = { country_event = { id = atb_ex.6 days = 2 } }

		# 远征军整体搬到银河边缘外的星团（首都是起源要塞的国家，每个银河最多一个）
		random_playable_country = {
			limit = {
				exists = capital_scope
				capital_scope = { has_planet_flag = atb_origin_fortress }
			}
			atb_ex_relocate_expedition = yes
		}

		# 每局必定生成天选民，固定在 7 点半方向：原版 1/4 概率的随机生成已在覆盖文件里关掉，
		# 这里用原版同一个初始化器在固定方位生成。3.10：等保底列强生成好（第二天，atb_ex.7），
		# 虫洞出口固定在大君主同盟领地的母星系
		random_country = { country_event = { id = atb_ex.7 days = 1 } }
	}
}

# 每年兜底（on_yearly_pulse）：新国家平时由 atb_ex.3 当场补上，这里只接漏网的——
# 主要是从旧版本存档读进来的国家（读档不会再触发 on_game_start）。
# 2.7 及以前这是每月一次，2.8 起改成每年。
event = {
	id = atb_ex.2
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		atb_ex_apply_galaxy_rules = yes
		atb_ex_grant_habitat_techs = yes
		# 3.6：旧存档补做虫洞与好感（做过就有全局旗标，atb_ex.6 自己会跳过）
		random_country = { country_event = { id = atb_ex.6 } }
		# 3.10：列强的恒星基地上限 +10 拆成单独的修正（大君主同盟领地不给），旧存档在这里补发
		every_country = {
			limit = {
				has_country_flag = atbgp_great_power
				NOT = { has_country_flag = atbgp_power_3 }
				NOT = { has_modifier = atbgp_starbase_bonus }
			}
			add_modifier = { modifier = atbgp_starbase_bonus days = -1 }
		}
		# 3.11：遥远之约定的 L 星群纳米星球可环境改造，旧存档在这里补（已有修正的星球会跳过）
		if = {
			limit = { any_country = { has_country_flag = atb_distant_promise } }
			atb_ex_lcluster_terraformable = yes
		}
		# 3.19：旧存档里还是指挥官的美铃、萃香、雏、米斯蒂娅转成行政官
		atb_th_convert_officials = yes
	}
}

# 新国家出现时当场补上银河规则与居住站科技（国家作用域）。
# on_country_created：create_country / create_rebels —— 叛军、被释放的附庸、中期刷出的天灾势力

}

# 3.6：无形链接 ↔ 逆鳞（远星同盟）、极限边缘 ↔ 神话（燃烧白鹿）两个天然虫洞（原版要求先研究「虫洞稳定」才能通行），
# 以及燃烧白鹿与远星同盟互相 +200 好感。开局第三天做一次，旧存档由每年的 atb_ex.2 补；各自做完设全局旗标。
# 虫洞写法同原版 central_crystal_events：在 A 星系里先让 prev（B）生成虫洞，再在 A 生成，然后 link_wormholes = prev
country_event = {
	id = atb_ex.6
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = atb_ex_wormholes_36 }
				exists = event_target:atb_rl_sys_preftl
				exists = event_target:atbgp_sys_2_1
				exists = event_target:atb_rl_sys_blackhole
				exists = event_target:atbgp_sys_1_5
			}
			set_global_flag = atb_ex_wormholes_36
			event_target:atbgp_sys_2_1 = {
				event_target:atb_rl_sys_preftl = {
					prev = {
						spawn_natural_wormhole = {
							bypass_type = wormhole
							random_pos = no
							orbit_angle = 360
						}
					}
					spawn_natural_wormhole = {
						bypass_type = wormhole
						random_pos = no
						orbit_angle = 360
					}
					link_wormholes = prev
				}
			}
			event_target:atbgp_sys_1_5 = {
				event_target:atb_rl_sys_blackhole = {
					prev = {
						spawn_natural_wormhole = {
							bypass_type = wormhole
							random_pos = no
							orbit_angle = 360
						}
					}
					spawn_natural_wormhole = {
						bypass_type = wormhole
						random_pos = no
						orbit_angle = 360
					}
					link_wormholes = prev
				}
			}
		}
		if = {
			limit = {
				NOT = { has_global_flag = atb_ex_deer_star_ties_36 }
				any_country = { has_country_flag = atbgp_power_1 }
				any_country = { has_country_flag = atbgp_power_2 }
			}
			set_global_flag = atb_ex_deer_star_ties_36
			random_country = {
				limit = { has_country_flag = atbgp_power_1 }
				save_event_target_as = atb_ex_deer
			}
			random_country = {
				limit = { has_country_flag = atbgp_power_2 }
				save_event_target_as = atb_ex_star
			}
			event_target:atb_ex_deer = {
				add_opinion_modifier = {
					who = event_target:atb_ex_star
					modifier = atb_ex_deer_star_ties
				}
			}
			event_target:atb_ex_star = {
				add_opinion_modifier = {
					who = event_target:atb_ex_deer
					modifier = atb_ex_deer_star_ties
				}
			}
		}
	}
}

# 3.10：开局第二天生成天选民（保底列强已经生成好，虫洞出口能固定在大君主同盟领地母星）
country_event = {

			limit = {
				has_first_contact_dlc = yes

		}

# 3.10：开门——天选民或大君主同盟领地的舰队第一次穿过伊索姆之门的虫洞（on_entering_wormhole）时，两国结盟。
# THIS = 舰队，FROM = 跳往的星系，FROMFROM = 出发的星系
fleet_event = {
	id = atb_ex.8
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_global_flag = atb_ex_chosen_pact }
		exists = event_target:chosen_empire
		exists = event_target:atb_ex_overlord
		exists = owner
		OR = {
			from = { has_star_flag = ithomes_gate }
			fromfrom = { has_star_flag = ithomes_gate }
		}
		owner = {
			OR = {
				is_same_value = event_target:chosen_empire
				is_same_value = event_target:atb_ex_overlord
			}
		}
	}

	immediate = {
		set_global_flag = atb_ex_chosen_pact
		event_target:chosen_empire = { atb_ex_chosen_ally_overlord = yes }
	}
}
