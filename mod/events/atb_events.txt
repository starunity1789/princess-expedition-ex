# 首都初始化（殖民地作用域，empire_init_capital_colony，排在原版 game_start.12 之后）
colony_event = {
	id = atb.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = owner
		owner = {
			OR = {
				has_origin = origin_atb_princess_expedition
				has_origin = origin_atb_distant_promise
				has_country_flag = atb_princess_expedition
			}
		}
	}

	immediate = {
		owner = { atb_swap_origin = yes }
		# 如果起源切换晚于原版首都初始化，原版不会做合成繁衍的首都设置，这里补上
		if = {
			limit = {
				NOT = { has_building = building_identity_repository }
			}
			owner = {
				set_variable = {
					which = num_dead_identities
					value = 300
				}
				give_technology = { tech = tech_powered_exoskeletons message = no }
			}
			add_building = building_abandoned_gene_clinic
			while = {
				count = 2
				add_deposit = d_abandoned_cities
			}
			add_building = building_identity_repository
			create_pop_group = {
				size = 900
				species = owner_main_species
			}
		}
		# 3.15：黄金世代的黄金学院（原版在首都初始化里按起源加；标记设得晚时这里补）
		if = {
			limit = {
				owner = { has_country_flag = atb_golden_generation }
				NOT = { has_building = building_golden_academy }
			}
			add_building = building_golden_academy
		}
		# 原版合成繁衍会把首都尺寸设为 20，这里恢复为 25
		set_planet_size = 25
		set_planet_flag = atb_origin_fortress
		# 起源要塞的三处特殊地形
		if = {
			limit = {
				NOT = { has_planet_flag = atb_fortress_deposits_added }
			}
			set_planet_flag = atb_fortress_deposits_added
			add_deposit = d_atb_fortress_core
			add_deposit = d_atb_star_sea_monument
			add_deposit = d_atb_sensor_array
		}
		if = {
			limit = {
				NOT = { has_planet_flag = atb_fortress_institute_added }
			}
			set_planet_flag = atb_fortress_institute_added
			add_deposit = d_atb_elemental_seal_institute
		}
	}
}

# 全局开局（on_game_start，排在原版 distar.8000 随机 L 星群结局之后）
# 有远征军在场时：L 星群结局固定为灰色风暴，并保证能找到小灰。
# 如果游戏规则屏蔽了破坏性内容（灰色风暴不会出现），则改为空星群结局，小灰按原版方式出现。
# 3.10：起源「遥远之约定」固定为空星群结局（没有灰色风暴，小灰的异常现象由开局布置保证出现）；
# 不再要求开启 L 星门（atb_ex_ensure_lgates 保证银河里至少 7 座）。
event = {
	id = atb.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_country = { has_country_flag = atb_princess_expedition }
	}

	immediate = {
		remove_global_flag = dragon_season
		remove_global_flag = gray_goo_empire_set
		if = {
			limit = { any_country = { has_country_flag = atb_distant_promise } }
			remove_global_flag = gray_goo_crisis_set
			remove_global_flag = active_gray_goo
		}
		else_if = {
			limit = { blocks_disruptive_gameplay = no }
			set_global_flag = gray_goo_crisis_set
			set_global_flag = active_gray_goo
			set_global_flag = atb_forced_gray_tempest
		}
		else = {
			remove_global_flag = gray_goo_crisis_set
			remove_global_flag = active_gray_goo
		}
	}
}

# 每月：锁肖像、锁职位、记录等级、检查是否需要复活
country_event = {
	id = atb.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = atb_princesses_installed
	}

	immediate = {
		atb_princess_maintenance = yes
		atb_revive_princesses = yes
		# 旧存档补发：开局之后才加入的内容（特殊地形、光耀者）
		every_owned_planet = {
			limit = {
				has_planet_flag = atb_origin_fortress
				NOT = { has_planet_flag = atb_fortress_deposits_added }
			}
			set_planet_flag = atb_fortress_deposits_added
			add_deposit = d_atb_fortress_core
			add_deposit = d_atb_star_sea_monument
			add_deposit = d_atb_sensor_array
		}
		# 2.5 新增的高位元素秘封研究院
		every_owned_planet = {
			limit = {
				has_planet_flag = atb_origin_fortress
				NOT = { has_planet_flag = atb_fortress_institute_added }
			}
			set_planet_flag = atb_fortress_institute_added
			add_deposit = d_atb_elemental_seal_institute
		}
		atb_start_luminary = yes
		# 合成黎明（原版合成繁衍事件链的终章 machine_age.3315）之后，解锁远征军体制
		if = {
			limit = {
				has_modifier = synthetic_dawn
				NOT = { has_country_flag = atb_regime_unlocked }
			}
			set_country_flag = atb_regime_unlocked
			country_event = { id = atb.40 days = 7 }
		}
		# 3.6：合成黎明之后（需要迷雾之影 DLC），半年到一年半内由二皇女开启灵能之路（atb.50）
		if = {
			limit = {
				has_modifier = synthetic_dawn
				has_shroud_dlc = yes
				NOT = { has_country_flag = atb_psionic_path_open }
				NOT = { has_country_flag = atb_psionic_call_scheduled }
			}
			set_country_flag = atb_psionic_call_scheduled
			country_event = { id = atb.50 days = 180 random = 360 }
		}
	}
}
