	title = forceofwill.5100.name
	desc = forceofwill.5100.desc
	is_triggered_only = yes
	picture = GFX_evt_golden_generation_students
	show_sound = event_golden_generation_students_talk

	event_chain = chain_golden_generation

	trigger = {
		atb_has_golden_generation = yes
		NOT = { has_country_flag = forceofwill.5100_event_triggered }
		count_leader_in_training = { count > 0 }
		resource_stockpile_compare = {
			resource = energy
			value >= 500
		}
	}

	immediate = {
		set_country_flag = forceofwill.5100_event_triggered
		random_owned_planet = {
			limit = { has_carrier_flag = golden_academy_built }
			save_event_target_as = golden_academy_planet
		}
		random_leader_in_training = {
			limit = {
				NOT = { has_leader_flag = golden_generation_had_trait_event }
			}
			save_event_target_as = selected_leader_in_training
			set_leader_flag = golden_generation_had_trait_event
		}
	}

	option = {
		name = forceofwill.5100.a
		allow = {
			resource_stockpile_compare = {
				resource = energy
				value >= 500
			}
		}
		add_resource = { energy = -500 }
		event_target:selected_leader_in_training = {
			switch = {
				trigger = leader_class
				commander = { add_trait_with_concept_tt = { TRAIT = leader_trait_restrained } }
				official = { add_trait_with_concept_tt = { TRAIT = leader_trait_principled } }
				scientist = { add_trait_with_concept_tt = { TRAIT = leader_trait_carefree } }
			}
			add_trait_chance_with_concept_tt = {
				CHANCE = 20
				TRAIT = leader_trait_arrested_development
			}
		}
		random_leader_in_training = {
			limit = {
				NOT = { is_same_value = event_target:selected_leader_in_training }
			}
			add_trait_chance_with_concept_tt = {
				CHANCE = 10
				TRAIT = leader_trait_sadistic
			}
		}
	}

	option = {
		name = forceofwill.5100.b
		allow = {
			resource_stockpile_compare = {
				resource = energy
				value >= 500
				mult = value:count_leaders_in_training
			}
		}
		add_resource = {
			energy = -500
			mult = value:count_leaders_in_training
		}
		every_leader_in_training = {
			limit = { always = yes }
			hidden_effect = {
				random = {
					chance = 15
					add_trait = { trait = leader_trait_arrested_development }
				}
			}
			if = {
				limit = { has_trait = leader_trait_resilient }
				if = {
					limit = { has_trait = leader_trait_adaptable }
					add_experience = 100
				}
				else = {
					add_trait_with_concept_tt = { TRAIT = leader_trait_adaptable }
				}
			}
			else = {
				add_trait_with_custom_concept_tt = { TRAIT = leader_trait_resilient }
			}
		}
		custom_tooltip = forceofwill.5100.a.tt
	}

	option = {
		name = OPTION_DENIED
		event_target:selected_leader_in_training = {
			add_trait_with_concept_tt = { TRAIT = leader_trait_adaptable }
			if = {
				limit = { has_trait = leader_trait_melancholic }
				if = {
					limit = { has_trait = leader_trait_stubborn }
					add_trait_with_custom_concept_tt = { TRAIT = leader_trait_destructive }
				}
				else = {
					add_trait_with_concept_tt = { TRAIT = leader_trait_stubborn }
				}
			}
			else = {
				add_trait_with_custom_concept_tt = { TRAIT = leader_trait_melancholic }
			}
		}
		default_hide_option = yes
	}
}

	title = forceofwill.5120.name

				limit = { exists = event_target:selected_leader_in_training_1 }

				text = forceofwill.5120.desc.leader_in_training_1
			}
			if = {
				limit = { exists = event_target:selected_leader_in_training_2 }
				text = newline
				text = forceofwill.5120.desc.leader_in_training_2
			}
			if = {
				limit = { exists = event_target:selected_leader_in_training_3 }
				text = newline
				text = forceofwill.5120.desc.leader_in_training_3
			}
			if = {
				limit = { exists = event_target:selected_leader_in_training_4 }
				text = newline
				text = forceofwill.5120.desc.leader_in_training_4
			}
			text = newline
			text = forceofwill.5120.desc.common_2
		}
	}
	is_triggered_only = yes
	picture = GFX_evt_golden_generation_classroom
	show_sound = event_golden_generation_students_talk

	event_chain = chain_golden_generation

	trigger = {
		atb_has_golden_generation = yes
		NOT = { has_country_flag = forceofwill.5120_event_triggered }
		NOT = {
			any_leader_in_training = {
				num_leader_traits = { value = 0 }
			}
		}
	}

	immediate = {
		set_country_flag = forceofwill.5120_event_triggered
		while = {
			count = value:count_leaders_in_training
			random_list = {
				1 = {
					modifier = {
						factor = 0
						exists = event_target:selected_leader_in_training_1
					}

						limit = { NOT = { has_leader_flag = selected_for_exp_event } }
						set_leader_flag = selected_for_exp_event

				}

					modifier = {
						factor = 0
						exists = event_target:selected_leader_in_training_2
					}
					random_leader_in_training = {
						limit = { NOT = { has_leader_flag = selected_for_exp_event } }
						set_leader_flag = selected_for_exp_event
						save_event_target_as = selected_leader_in_training_2
					}
				}
				1 = {
					modifier = {
						factor = 0
						exists = event_target:selected_leader_in_training_3
					}
					random_leader_in_training = {
						limit = { NOT = { has_leader_flag = selected_for_exp_event } }
						set_leader_flag = selected_for_exp_event
						save_event_target_as = selected_leader_in_training_3
					}
				}
				1 = {
					modifier = {
						factor = 0
						exists = event_target:selected_leader_in_training_4
					}
					random_leader_in_training = {
						limit = { NOT = { has_leader_flag = selected_for_exp_event } }
						set_leader_flag = selected_for_exp_event
						save_event_target_as = selected_leader_in_training_4
					}
