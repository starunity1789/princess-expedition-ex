# 皇女的远征军EX 3.0：AI 自动星球的 6 座建筑（都放在城市区划的政府分区里，正好 6 个格）
# 产出 = 系数 × 星球变量 atb_ai_auto_jobs（A，虚拟岗位数）。系数见 common/scripted_variables/atb_ai_auto_variables.txt。
# 3.0 起的基准：1 点 A 相当于一个原版岗位，按份额折算：
#   科研 30%（研究员 3/岗）  合金 15%（冶金工 3/岗）  能量 15%（技术员 6/岗）
#   团结 10%（官僚 3/岗）    消费品 10%（工匠 6/岗）  矿物 5%（矿工 4/岗）  食物 5%（农民 6/岗）
#   另有少量贸易与战略资源；不用消费品的帝国（格式塔）把那 10% 改成合金与科研
# 产出挂在原版岗位类别下，AI 自己的科技与传统加成照常生效。没有维护费。
# 不能建造、不能拆除；星球失去 atb_ai_auto 标记（换给玩家）时自动消失。

@atb_ai_auto_buildtime = 1

# 自治中枢：首府 + 科研 + 全部星球修正 + 海军容量
building_atb_ai_core = {
	base_buildtime = @atb_ai_auto_buildtime
	capital = yes
	can_build = no
	can_demolish = no
	can_be_ruined = no
	can_be_disabled = no
	position_priority = 0
	capital_tier = 10
	exempt_from_ai_planet_specialization = yes
	category = government
	icon = building_capital

	building_sets = {
		government
	}

	potential = {
		exists = owner
		has_carrier_flag = atb_ai_auto
	}

	destroy_trigger = {
		NOT = { has_carrier_flag = atb_ai_auto }
	}

	# FTL 抑制器（同原版要塞）
	planetary_ftl_inhibitor = yes

	planet_modifier = {
		planet_housing_add = 2000
		planet_amenities_add = 2000
		planet_stability_add = 25
		planet_crime_add = -100
		# 行星护盾（同原版行星护盾发生器）
		planet_orbital_bombardment_damage = -0.5
		# 10 支防御陆军
		planet_defense_armies_add = 10
	}

	# 每个人口一个岗位，永不失业
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = no }
		}
		job_atb_ai_auto_worker_per_pop = 1
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = yes }
		}
		job_atb_ai_auto_drone_per_pop = 1
	}

	triggered_country_modifier = {
		potential = { exists = owner }
		country_naval_cap_add = @atb_ai_auto_naval_cap
		mult = root.planet.atb_ai_auto_jobs
	}

	resources = {
		category = planet_researchers
		produces = {
			physics_research = @atb_ai_auto_out_research
			society_research = @atb_ai_auto_out_research
			engineering_research = @atb_ai_auto_out_research
			multiplier = value:atb_ai_auto_output
		}
		produces = {
			trigger = { owner? = { country_uses_consumer_goods = no } }
			physics_research = @atb_ai_auto_out_research_gestalt
			society_research = @atb_ai_auto_out_research_gestalt
			engineering_research = @atb_ai_auto_out_research_gestalt
			multiplier = value:atb_ai_auto_output
		}
	}
}
