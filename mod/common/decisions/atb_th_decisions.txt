# 招募幻想乡来客：玩家首都决议，按职业分三个，花食物随机招募一位还没有的该职业东方角色，直到招完。
# 这是东方国策之外唯一的获得途径；AI 看不到。

# 500 食物：随机招募一位还没有的东方科学家（7 位）
decision_atb_th_recruit_scientist = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = decision_luxuries

	resources = {
		category = decisions
		cost = {
			food = 500
		}
	}

	potential = {
		is_capital = yes
		owner = {
			is_ai = no
			atb_th_has_unrecruited_scientist = yes
		}
	}

	effect = {
		custom_tooltip = decision_atb_th_recruit_scientist_tt
		hidden_effect = {
			owner = {
				atb_th_recruit_scientist = yes
			}
		}
	}

	ai_weight = {
		weight = 0
	}
}
