# Stealth overlay applied after the base set_ai_ship_roles in every fleet doctrine.
# Sneak attackers with First Contact reassign their cloak-capable sizes (corvette and destroyer
# classes) to stealth roles. Cloaking tech is not checked, so designs need no regeneration when it lands.
if = {
	limit = {
		has_first_contact_dlc = yes
		has_ai_personality_behaviour = sneak_attacker
	}
	# Roles are parameters so each doctrine can pick its own stealth roles per size. Engine inline
	# scripts have no default values, so every caller must pass all of these parameters.
	set_ai_ship_roles = {
		# Regular Ships
		corvette = { $CORVETTE$ = 60 screen = 40 } # Stellar AI (vanilla $CORVETTE$): keeps point defence screens; vanilla replaces the whole hull role, and these empires had none
		frigate				= $FRIGATE$
		destroyer = { $DESTROYER$ = 60 screen = 40 } # Stellar AI (vanilla $DESTROYER$): keeps point defence screens; vanilla replaces the whole hull role, and these empires had none
		# Galactic Nemesis Ships
		crisis_corvette		= $CRISIS_CORVETTE$
		crisis_destroyer	= $CRISIS_DESTROYER$
		# Nanotech Ships
		nanite_swarmer		= $NANITE_SWARMER$
		# Biological Ships
		mauler_stage_1 = { $MAULER$ = 60 screen = 40 } # Stellar AI (vanilla $MAULER$): keeps point defence screens; vanilla replaces the whole hull role, and these empires had none
		weaver_stage_1		= $WEAVER$
	}
}
