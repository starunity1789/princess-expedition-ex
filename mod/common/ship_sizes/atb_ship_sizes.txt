# 皇女的远征军：无畏舰
# 战列舰的船体（模型、舰段、推进器、战斗电脑、舰名都沿用战列舰），舰首可以装泰坦级（T 槽）武器，使用泰坦反应堆。
# 组件归属由 common/scripted_triggers/07_scripted_triggers_ships.txt 里的 ship_uses_* 触发器决定。
# 所有国家（包括 AI）研究 tech_atb_dreadnought 后都能建造。

atb_dreadnought = {
	entity = battleship_entity
	use_shipnames_from = battleship
	use_ai_design_role_from = battleship

	formation_priority = @battleship_formation_priority
	max_speed = @speed_slow
	acceleration = 0.2
	rotation_speed = 0.15
	collision_radius = @battleship_collision_radius
	max_hitpoints = 6000
	modifier = {
		ship_evasion_add = @battleship_evasion
	}
	size_multiplier = @battleship_size_multiplier
	map_counter_icon = ship_counter_32
	icon = ship_size_military_8
	fleet_slot_size = 4
	section_slots = { "bow" = { locator = "part1" } "mid" = { locator = "part2" } "stern" = { locator = "part3" } }
	num_target_locators = 4
	is_space_station = no
	base_buildtime = 600
	can_have_federation_design = yes
	enable_default_design = yes
	enable_3dview_in_ship_browser = yes

	default_behavior = artillery

	role_background = "GFX_role_selection_battleship"
	ship_roles = { artillery carrier gunship }

	prerequisites = { "tech_atb_dreadnought" }

	combat_disengage_chance = 1.25

	class = shipclass_military
	construction_type = starbase_shipyard
	required_component_set = "power_core"
	required_component_set = "ftl_components"
	required_component_set = "thruster_components"
	required_component_set = "sensor_components"
	required_component_set = "combat_computers"

	potential_country = {
		country_uses_bio_ships = no
	}

	resources = {
		category = ships
		upkeep = {
			energy = 10
			alloys = 5
		}
		logistics = {
			trade = @battleship_logistic_upkeep_trade
		}
	}

	min_upgrade_cost = {
		alloys = 12
	}

	ai_ship_data = {
		fraction = {
			base = 0
			modifier = {
				add = 30
				uses_standard_ship_sizes = yes
			}
		}
	}
}
