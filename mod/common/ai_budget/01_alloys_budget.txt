alloys_expenditure_ships = {
	resource = alloys
	type = expenditure
	category = ships

	potential = {
		always = yes
	}

	weight = {
		weight = 1.0

		# War mobilisation replaces, rather than multiplies, the routine rebuild boost.
		modifier = {
			factor = @stellarai_war_ship_budget_factor
			stellarai_war_fleet_push = yes
		}

		# Sustain war spending when the stronger rebuilding priority is not active.
		modifier = {
			factor = 3
			OR = {
				recently_lost_war = yes
				is_at_war = yes
			}
			NOT = { stellarai_war_fleet_push = yes }
			NOT = { stellarai_fleet_readiness_focus = yes }
		}

		# Preserve the established capacity limits without choking wartime rebuilding.
		modifier = {
			factor = 0.65
			used_naval_capacity_percent >= 1.0
			is_at_war = no
		}
		modifier = {
			factor = 0.82
			used_naval_capacity_percent >= 1.15
			is_at_war = yes
		}

		# Biological Ships use less alloys
		modifier = {
			factor = 0.33
			country_uses_bio_ships = yes
		}

		modifier = {
			factor = 5
			used_naval_capacity_percent < 1.5
			OR = {
				any_neighbor_country = {
					has_ascension_perk = ap_become_the_crisis
				}
				any_country = {
					is_crisis_faction = yes
				}
			}
		}

		# Ordinary fleet rebuilding needs no technology lead or intelligence.
		# A verified science lead can also activate this single, non-stacking priority.
		modifier = {
			factor = 4.0
			NOT = { stellarai_war_fleet_push = yes }
			stellarai_fleet_readiness_focus = yes
		}
		modifier = {
			factor = 0.60
			is_at_war = no
			stellarai_economic_crisis = yes
		}
		# Expansion trade-off: while real colony targets remain and the empire is small, safe and at
		# peace, alloys go to colony ships first. It ends by itself when targets or space run out.
		modifier = {
			factor = 0.12
			stellarai_expansion_phase = yes
			is_machine_empire = no
			NOT = { stellarai_militant_pressure = yes }
			NOT = { stellarai_opportunity_ready = yes }
		}
		modifier = {
			factor = 0.30
			stellarai_expansion_phase = yes
			is_machine_empire = yes
			NOT = { has_valid_civic = civic_machine_terminator }
			NOT = { stellarai_militant_pressure = yes }
			NOT = { stellarai_opportunity_ready = yes }
		}
		modifier = {
			factor = 0.60
			stellarai_expansion_phase = yes
			OR = {
				has_valid_civic = civic_machine_terminator
				stellarai_militant_pressure = yes
			}
			NOT = { stellarai_opportunity_ready = yes }
		}
		modifier = {
			factor = 0.45
			is_at_war = no
			stellarai_machine_energy_emergency = yes
		}
		modifier = {
			factor = 0.85
			is_at_war = no
			stellarai_machine_minerals_emergency = yes
		}
		modifier = {
			factor = 1.15
			stellarai_machine_basic_econ_stable = yes
			used_naval_capacity_percent < 1.0
		}
	}

	desired_min = {
		base = 0

		modifier = {
			add = @ai_ship_budget_desired_min_per_year
			country_uses_bio_ships = no
		}

		modifier = {
			factor = trigger:years_passed
		}
	}
}

	potential = {

			AND = {
				is_at_war = no

				}

		}
	}

	weight = {
		weight = 0.2

		# afford them, even without a verified technology or fleet advantage.

			factor = 0.25
			is_at_war = no
			used_naval_capacity_percent < 0.60
			stellarai_fleet_readiness_focus = yes
		}
		modifier = {
			factor = 1.50
			is_at_war = no
			has_monthly_income = { resource = alloys value > 25 }
			resource_stockpile_compare = { resource = alloys value > 1200 }
			OR = {
				stellarai_research_ready = yes
				stellarai_opportunity_ready = yes
			}
		}

		# spend more on ship upgrades if at capacity
		modifier = {
			factor = 2.5
			is_at_war = no
			used_naval_capacity_percent >= 1.0
		}

		# Biological Ships use less alloys
		modifier = {
			factor = 0.33
			country_uses_bio_ships = yes
		}
	}

alloys_expenditure_starbases = {
	resource = alloys
	type = expenditure
	category = starbases

	potential = {
		NOT = {
			is_country_type = fallen_empire
		}
	}

	weight = {
		weight = 0.4

		# spend more on starbase upgrades if at fleet cap
		modifier = {
			factor = 2
			used_naval_capacity_percent >= 1.0
		}

		# Reduce spending on claiming new systems if we have unclaimed planets
		modifier = {
			is_machine_empire = yes
			has_country_resource = { type = alloys amount < 450 }
			factor = 0.25 # more aggresive saving for robots since their colony ships cost more
			ai_colonize_plans > 0
		}
		# Reduce spending on claiming new systems if we have unclaimed planets
		modifier = {
			is_machine_empire = no
			has_country_resource = { type = alloys amount < 250 }
			factor = 0.50
			ai_colonize_plans > 0
		}

		modifier = {
			factor = 1.5
			used_starbase_capacity_percent < 1
			any_owned_starbase = {
				can_be_upgraded = yes
			}
		}
		modifier = {
			factor = 1.5
			OR = {
				any_owned_starbase = {
					NOT = { has_starbase_size = starbase_outpost }
					can_be_upgraded = yes
				}
				any_owned_nonprimary_starbase = {
					can_be_upgraded = yes
				}
			}
		}

		# Biological Starbases use less alloys
		modifier = {
			factor = 0.5
			country_uses_bio_ships = yes
		}

		# Nomads rely on Waystations. Prioritize starbase spending.
		modifier = {
			factor = 2
			is_nomadic = yes
		}
	}

	# Upgrade an arkship the empire already owns before it builds another one. The floor releases as soon as
	# the upgrade is queued, because has_upgradable_arkship goes false then.
	desired_min = {
		base = 0
		modifier = {
			add = @arkship_cost_alloys_5000
			is_nomadic = yes
			has_upgradable_arkship = yes
		}
	}

	desired_max = {
		base = 500
		modifier = {
			# Allow saving enough for a Starhold (1000 alloys)
			add = 500
			country_uses_bio_ships = no
			has_technology = tech_starbase_3
			any_owned_starbase = {
				has_starbase_size = starbase_starport
				can_be_upgraded = yes
			}
		}
		# Bioship Starhold costs 500 alloys - base 500 is sufficient, no add needed
		modifier = {
			# Allow saving enough for a Starfortress (2500 alloys)
			add = 2000
			country_uses_bio_ships = no
			has_technology = tech_starbase_4
			any_owned_starbase = {
				has_starbase_size = starbase_starhold
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for a Bioship Starfortress (1250 alloys)
			add = 750
			country_uses_bio_ships = yes
			has_technology = tech_starbase_4
			any_owned_starbase = {
				has_starbase_size = starbase_starhold
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for a Citadel (5000 alloys)
			add = 4500
			country_uses_bio_ships = no
			has_technology = tech_starbase_5
			any_owned_starbase = {
				has_starbase_size = starbase_starfortress
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for a Bioship Citadel (2500 alloys)
			add = 2000
			country_uses_bio_ships = yes
			has_technology = tech_starbase_5
			any_owned_starbase = {
				has_starbase_size = starbase_starfortress
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for an Arkship tier 2 upgrade (5000 alloys)
			# Only one tier upgrade can be bought at a time, so don't stack with the tier 3 reservation
			add = 5000
			is_nomadic = yes
			has_technology = tech_arkship_tier_2
			NOT = {
				has_technology = tech_arkship_tier_3
			}
			has_upgradable_arkship = yes
		}
		modifier = {
			# Allow saving enough for an Arkship tier 3 upgrade (5000 alloys)
			add = 5000
			is_nomadic = yes
			has_technology = tech_arkship_tier_3
			has_upgradable_arkship = yes
		}
		modifier = {
			# Allow saving enough for the Exodus Jump spinal module (5000 alloys)
			add = 5000
			is_nomadic = yes
			has_technology = tech_arkship_exodus_jump
			any_owned_fleet = {
				starbase? = {
					arkship_spinal_slot_free = yes
				}
			}
		}
		modifier = {
			add = 5000
			has_ascension_perk = ap_cosmogenesis
		}
	}
}
