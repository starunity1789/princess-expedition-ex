shift_ethic_from_embrace = {
	if = {
		limit = {
			exists = ideology
			is_fanatic_$ETHIC$ = yes
			prev = { is_faction_fanatic = yes }
		}
		if = {
			limit = { ideology = { is_spiritualist = yes } }
			enable_special_project = {
				name = "ABSOLUTE_ETHIC_PROJECT_SPIRITUALIST"
				owner = this
			}
		}
		else = {
			enable_special_project = {
				name = "ABSOLUTE_ETHIC_PROJECT"
				owner = this
			}
		}
	}
	else_if = {
		limit = { is_fanatic_$ETHIC$ = no }
		hidden_effect = {
			# Remember which fanatic factions are ethic-aligned before the shift
			every_pop_faction = {
				limit = {
					is_faction_fanatic = yes
					is_$ETHIC$ = no
					owner_follows_faction_ethic = yes
				}
				if = {
					limit = { owner_follows_faction_ethic_fanatic = yes }
					set_pop_faction_flag = ethics_shift_was_fanatic_aligned
				}
				else = {
					set_pop_faction_flag = ethics_shift_was_aligned
				}
			}
		}
		shift_ethic = ethic_$ETHIC$
		create_message = {
			type = MESSAGE_FACTION_EMBRACED_ETHIC_SHIFTED
			localization = MESSAGE_FACTION_EMBRACED_ETHIC_SHIFTED_DESC
			days = 100
			onclick = government
		}
		hidden_effect = {
			# A fanatic faction whose ethic was downgraded or removed by the shift reacts
			ordered_pop_faction = {
				limit = {
					OR = {
						AND = {
							has_pop_faction_flag = ethics_shift_was_fanatic_aligned
							owner_follows_faction_ethic_fanatic = no
						}
						AND = {
							has_pop_faction_flag = ethics_shift_was_aligned
							owner_follows_faction_ethic = no
						}
					}
				}
				order_by = trigger:support
				position = 0
				prev = {
					country_event = {
						id = forceofwill.7555
						scopes = { from = prev }
					}
				}
			}
			every_pop_faction = {
				remove_pop_faction_flag = ethics_shift_was_fanatic_aligned
				remove_pop_faction_flag = ethics_shift_was_aligned
			}
		}
	}
	hidden_effect = { prev = { remove_pop_faction_flag = ai_triggered_embrace } }
}

generate_ideology = {
	set_country_flag = ideology_initiated # Just to make sure
	if = {
		limit = { is_ai = yes }
		set_country_flag = cannot_codify_ideology
	}
	create_ideology = {
	[[ETHIC]
		ethic = $ETHIC$
	]
	[[FACTION]
		faction = $FACTION$
	]
	[[NAME]
		name = $NAME$
	]
	[[TYPE]
		type = $TYPE$
	]
	[[TENETS]
		tenets = { $TENETS$ }
	]
	[[TENETS_FROM]
		tenets = $TENETS_FROM$
	]
		effect = {
			save_event_target_as = generated_ideology
			create_message = {
				type = MESSAGE_IDEOLOGY_EMERGED
				localization = MESSAGE_IDEOLOGY_EMERGED_DESC
				days = 30
				onclick = ideologies
			}
		}
	}
}

generate_ideology_from_faction = {
	owner? = {
		generate_ideology = { FACTION = prev }
	}
}

# Found an Ideology from one of the empire's fanatic factions, or from its own ethics when it has no fanatic faction
# scope = country
# ETHIC = <ethic key without its ethic_ prefix>, to found an Ideology of that ethic when the empire supports it (optional)
generate_ideology_from_faction_or_ethic = {
	[[ETHIC]
		random_pop_faction = {
			limit = {
				is_faction_fanatic = yes
				is_$ETHIC$ = yes
				NOT = { exists = ideology }
			}
			generate_ideology_from_faction = yes
		}
		if = {
			limit = {
				NOT = { exists = ideology }
				is_fanatic_$ETHIC$ = yes
			}
			generate_ideology = { ETHIC = ethic_fanatic_$ETHIC$ }
		}
	]
	if = {
		limit = { NOT = { exists = ideology } }
		random_pop_faction = {
			limit = {
				is_faction_fanatic = yes
				NOR = {
					exists = ideology
					has_ethic = ethic_gestalt_consciousness
				}
			}
			generate_ideology_from_faction = yes
		}
	}
	if = {
		limit = { NOT = { exists = ideology } }
		generate_ideology = yes
	}
}

# scope = country
generate_ideology_for_empire = {
	if = {
		limit = { is_ai = no }
		generate_ideology_from_faction_or_ethic = yes
	}
	else_if = {
		limit = {
			OR = {
				has_ai_personality = evangelising_zealots
				has_ai_personality = spiritual_seekers
			}
		}
		generate_ideology_from_faction_or_ethic = { ETHIC = spiritualist }
	}
	else_if = {
		limit = { has_ai_personality = honorbound_warriors }
		generate_ideology_from_faction_or_ethic = { ETHIC = militarist }
	}
	else_if = {
		limit = { has_ai_personality = democratic_crusaders }
		generate_ideology_from_faction_or_ethic = { ETHIC = egalitarian }
	}
	else_if = {
		limit = { has_ai_personality = fanatic_befrienders }
		generate_ideology_from_faction_or_ethic = { ETHIC = xenophile }
	}
	else = {
		generate_ideology_from_faction_or_ethic = yes
	}
}

		}

		}
	}
}

embrace_faction_effect = {
	add_modifier = { modifier = embraced_faction days = 3600 }
	owner = {
		set_country_flag = has_embraced_faction
		shift_ethic_from_embrace = { ETHIC = $ETHIC$ }
		hidden_effect = {
			add_modifier = { modifier = embraced_faction_timer days = 3600 }
			every_pop_faction = {
				limit = {
					NOT = { is_same_value = root }
				}
				if = {
					limit = { is_$ETHIC$ = yes }
					add_modifier = { modifier = embraced_faction days = 3600 }
				}
				else = {
					add_modifier = { modifier = embraced_another_faction days = 3600 }
				}
			}
		}
	}
	hidden_effect = {
		save_event_target_as = TargetFaction
		owner = {
			every_relation = {
				limit = {
					is_ai = no
					is_country_type = default
					has_communications = prev
					has_intel_level = {
						who = prev
						category = government
						level >= 1
					}
				}
				country_event = { id = factions.2000 }
			}
		}
	}
}

	}
}

start_abandon_ideology_process = {
	# Any Gestalt AI abandon conditions pending on this Ideology no longer apply
	remove_country_flag = ideology_abandon_strength_met
	remove_country_flag = ideology_abandon_deviancy_met

	# Add malus modifier
	add_modifier = {
		modifier = abandoning_ideology
		days = -1
	}
	# Add malus flag
	custom_tooltip = start_abandon_ideology
	set_country_flag = abandoning_ideology

	every_pop_faction = {
		limit = {
			ideology? = {
				is_same_value = prevprev.ideology
			}
		}
		faction_action = "cancel_promote_faction"
		faction_action = "suppress_faction"
	}
}

stop_abandon_ideology_process = {
	remove_modifier = abandoning_ideology
	custom_tooltip = stop_abandon_ideology
	hidden_effect = {
		remove_country_flag = abandoning_ideology
		remove_country_flag = ai_auto_abandoning_ideology

		every_ideology = {
			limit = { prev = { has_country_flag = adopting_ideology_from_invitation_@prev } }
			prev = {
				remove_country_flag = post_abandon_should_adopt_ideology_@prev
				remove_country_flag = adopting_ideology_from_invitation_@prev
			}
		}
		every_playable_country = {
			limit = { has_country_flag = invited_to_our_ideology_@prev }
			remove_country_flag = invited_to_our_ideology_@prev
		}

		every_pop_faction = {
			limit = {
				ideology? = {
					is_same_value = prevprev.ideology
				}
			}
			faction_action = "cancel_suppress_faction"
		}
	}
}

calc_highest_ideology_pops_planet = {
	set_variable = {
		which = highest_ideological_pops_count
		value = 0
	}
	every_owned_planet = {
		set_variable = {
			which = ideological_pops_count
			value = value:pop_ideology_amount_value|IDEOLOGY|fromfrom.root|WEIGHT|1|
		}
		if = {
			limit = {
				is_capital = no
				check_variable_arithmetic = {
					which = ideological_pops_count
					value > prev.highest_ideological_pops_count
				}
			}
			save_event_target_as = highest_ideology_pops_planet
			prev = {
				set_variable = {
					which = highest_ideological_pops_count
					value = prev.ideological_pops_count
				}
			}
		}
		clear_variable = ideological_pops_count
	}
	clear_variable = highest_ideological_pops_count
}

# This: the revolt situation.
# Deduces the organic founder for an ideological secession: a weighted-random organic species on the planet, favoring the most populous.
deduce_ideological_rebel_founder = {
	event_target:rebel_planet = {
		random_owned_species = {
			limit = {
				is_sapient = yes
				# Search for organic species
				OR = {
					is_archetype = BIOLOGICAL
					is_archetype = LITHOID
				}
			}
			weights = {
				base = 1
				complex_trigger_modifier = { # favor the species with the most pops on the planet
					trigger = count_owned_pop_amount
					trigger_scope = event_target:rebel_planet
					parameters = {
						limit = { is_same_species = prevprev } # prevprev = the candidate species
					}
					mode = add
				}
			}
			save_event_target_as = rebellion_species
		}
	}
}

# This: Planet scope
create_rebels_and_secede_planet = {
	deduce_rebel_planet_and_species = yes

	event_target:revolt_situation = {
		owner = {
			save_event_target_as = revolt_country
		}
		random_country = {
			limit = {
				has_country_flag = revolt_joiner_empire_@prev
				#Re-pick the backer with the trigger this chain polled its backers with.
				OR = {
					AND = {
						prev = { is_situation_type = situation_ideological_revolt }
						country_can_back_ideological_revolt_trigger = {
							SITUATION_OWNER = event_target:revolt_situation.owner
						}
					}
					AND = {
						prev = {
							NOT = { is_situation_type = situation_ideological_revolt }
						}
						separatism_joining_eligibility_trigger = {
							SPECIES = event_target:rebellion_species
							SITUATION = event_target:revolt_situation
						}
					}
				}
			}
			save_event_target_as = new_owner_empire
		}
	}

	create_rebels = {
		name = random
		ethos = random
		authority = random
		civics = random
		origin = "origin_separatists"
		species = event_target:rebellion_species
		flag = random
		nomadic = owner # a seceding Arkship spawns a nomadic empire
		effect = {
			[[DONT_COPY_IDEOLOGY]
				setup_separatist_rebel_empire = {
					DONT_COPY_IDEOLOGY = $DONT_COPY_IDEOLOGY$
				}
			]
			[[!DONT_COPY_IDEOLOGY]
				setup_separatist_rebel_empire = yes
			]
			event_target:rebel_planet = {
				secede_planet_to_rebels = yes
			}
			event_target:revolt_country = {
				every_owned_planet = { #Flip the additional joiner planets
					limit = {
						has_modifier = instability_join_revolt
						planet_stability < @[ stabilitylevel2 + 10 ]
						OR = {
							NOT = { exists = starbase }
							starbase = { is_owned_by = event_target:revolt_country }
						}
						has_ground_combat = no
						has_orbital_bombardment = no
					}
					if = { #if you check this in the original limit of every_owned_planet, it calcs it before the loop is executed. Which means it will set the owner of the system twice if there are two colonies in the system.
						limit = {
							exists = controller
							controller = { is_same_value = event_target:revolt_country }
						}
						remove_modifier = instability_join_revolt
						set_owner = event_target:rebel_empire
						if = {
							limit = {
								event_target:revolt_situation = { has_situation_flag = slave_revolt }
							}
							add_modifier = {
								modifier = free_at_last
								clear_on_owner_change = yes
								days = 7200
							}
						}
						else = {
							add_modifier = {
								modifier = insurgents_free
								clear_on_owner_change = yes
								days = 7200
							}
						}

						solar_system = {
							# Nomads have no primary starbases and cannot claim systems.
							if = {
								limit = {
									event_target:revolt_country = { is_nomadic = no }
								}
								if = {
									limit = { exists = starbase }
									starbase = { set_owner = event_target:rebel_empire }
								}
								add_claims = {
									who = event_target:revolt_country
									show_notification = no
								}
							}
							every_system_planet = {
								limit = {
									is_colony = yes
									NOT = { is_same_value = prevprev }
									is_owned_by = event_target:revolt_country
								}
								set_owner = event_target:rebel_empire
							}
							every_fleet_in_system = {
								limit = {
									is_owned_by = event_target:revolt_country
									can_go_mia = yes
								}
								set_mia = mia_return_home
							}
						}
					}
				}
			}
			every_system_within_border = { #Add a few systems
				every_neighbor_system = {
					limit = {
						exists = owner
						owner = { is_same_value = event_target:revolt_country }
						exists = starbase
						NOR = {
							starbase = {
								OR = {
									is_starbase_type = sfortress
									NOT = { is_owned_by = event_target:revolt_country }
								}
							}
							any_system_planet = {
								is_colony = yes
							}
						}
					}
					starbase = { set_owner = event_target:rebel_empire }
					add_claims = {
						who = event_target:revolt_country
						show_notification = no
					}
					every_fleet_in_system = {
						limit = {
							is_owned_by = event_target:revolt_country
							can_go_mia = yes
						}
						set_mia = mia_return_home
					}
				}
			}
			while = { #Get rid of bordergore systems
				limit = { #Note: system_within_border caching is quite funky at a time like this when many borders are changing hands. The starbase ownership is checked to be extra sure, since it is not cached (unlike system ownership - though that probably would have worked too)
					event_target:revolt_country = {
						any_system_within_border = {
							exists = starbase
							starbase.owner = { is_same_value = prevprev }
							count_neighbor_system = {
								count >= 2
								limit = {
									exists = starbase.owner
									starbase.owner = {
										is_same_value = event_target:rebel_empire
									}
								}
							}
							NOR = {
								starbase = { is_starbase_type = sfortress }
								any_system_planet = {
									is_colony = yes
								}
							}
						}
					}
				}
				event_target:revolt_country = {
					every_system_within_border = {
						limit = {
							exists = starbase
							starbase.owner = { is_same_value = prevprev }
							count_neighbor_system = {
								count >= 2
								limit = {
									exists = starbase.owner
									starbase.owner = {
										is_same_value = event_target:rebel_empire
									}
								}
							}
							NOR = {
								starbase = { is_starbase_type = sfortress }
								any_system_planet = {
									is_colony = yes
								}
							}
						}
						starbase = {
							set_owner = event_target:rebel_empire
						}
						add_claims = {
							who = event_target:revolt_country
							show_notification = no
						}
						every_fleet_in_system = {
							limit = {
								is_owned_by = event_target:revolt_country
								can_go_mia = yes
							}
							set_mia = mia_return_home
						}
					}
				}
			}

			if = {
				# The backer cannot own the seceded colonies (Nomad backer or Arkship target):
				# the rebels stay independent and become the backer's subject instead.
				limit = {
					exists = event_target:new_owner_empire
					ideological_revolt_creates_vassal_trigger = {
						BACKER = event_target:new_owner_empire
						TARGET = event_target:rebel_planet
					}
				}
				ruler = {
					save_event_target_as = exiled_rebel # not exiled here; unrest.4220 shows them as the rebel spokesperson
				}
				set_subject_of = {
					who = event_target:new_owner_empire
					preset = preset_nomad_tributary
				}
				event_target:new_owner_empire = {
					country_event = { id = unrest.4220 }
				}
			}
			else_if = {
				limit = { exists = event_target:new_owner_empire }
				set_country_flag = silence_destroy_event
				ruler = {
					save_event_target_as = exiled_rebel
					exile_leader_as = exiled_rebel
				}
				every_owned_starbase = {
					set_owner = event_target:new_owner_empire
				}
				every_owned_planet = {
					set_owner = event_target:new_owner_empire
					if = {
						limit = {
							event_target:revolt_situation = { has_situation_flag = slave_revolt }
						}
						add_modifier = {
							modifier = slaves_defected
							clear_on_owner_change = yes
							days = 7200
						}
					}
				}
				event_target:new_owner_empire = {
					country_event = { id = unrest.4220 }
				}
			}
			else = {
				add_resource = {
					minerals = 15000
					energy = 15000
					food = 15000
					consumer_goods = 10000
					alloys = 10000
					rare_crystals = 1000
					volatile_motes = 1000
					exotic_gases = 1000
					influence = 1000
					unity = 10000
				}
				#It copies the traditions and technologies in code

				create_fleet_from_naval_cap = 0.8
				last_created_fleet = {
					set_location = event_target:rebel_planet
				}
			}
		}
	}
}

		add_modifier = { modifier = $MODIFIER_PREFIX$_materialist_envoy }
	}
	else_if = {
		limit = { $ENVOY_COUNTRY$.ideology = { is_spiritualist = yes } }
		add_modifier = { modifier = $MODIFIER_PREFIX$_spiritualist_envoy }
	}
	else_if = {
		limit = { $ENVOY_COUNTRY$.ideology = { is_authoritarian = yes } }
		add_modifier = { modifier = $MODIFIER_PREFIX$_authoritarian_envoy }
	}
	else_if = {
		limit = { $ENVOY_COUNTRY$.ideology = { is_egalitarian = yes } }
		add_modifier = { modifier = $MODIFIER_PREFIX$_egalitarian_envoy }
	}
	else_if = {
		limit = { $ENVOY_COUNTRY$.ideology = { is_militarist = yes } }
		add_modifier = { modifier = $MODIFIER_PREFIX$_militarist_envoy }
	}
	else_if = {
		limit = { $ENVOY_COUNTRY$.ideology = { is_pacifist = yes } }
		add_modifier = { modifier = $MODIFIER_PREFIX$_pacifist_envoy }
	}
	else_if = {

		add_modifier = { modifier = $MODIFIER_PREFIX$_xenophile_envoy }
	}

try_change_disruption_effect_multiplier = {
	if = {
		limit = {
			NOT = { is_variable_set = disruption_effect_multiplier }
		}
		set_variable = {
			which = disruption_effect_multiplier
			value = 0
		}
	}

	if = {
		limit = {
			check_variable = {
				which = disruption_effect_multiplier
				value != $VALUE$
			}
		}
		remove_modifier = ideological_disruption
		remove_modifier = moderate_ideological_disruption
		remove_modifier = severe_ideological_disruption
		every_owned_planet = {
			remove_modifier = ideological_momentum_low_stability
			remove_modifier = ideological_momentum_moderately_low_stability
			remove_modifier = ideological_momentum_dangerously_low_stability
			remove_modifier = ideological_momentum_low_gestalt
			remove_modifier = ideological_momentum_moderately_low_gestalt
			remove_modifier = ideological_momentum_dangerously_low_gestalt
		}

		set_variable = {
			which = disruption_effect_multiplier
			value = $VALUE$
		}
	}
}

check_colony_ethic_shift_message = {
	if = {
		limit = { owner = { is_$ETHIC$ = no } }
		if = { # TODO AUG-55014 - Colony message targets are not implemented
			limit = { is_planet_class = pc_ark }
			fleet = { save_event_target_as = ethic_shift_message_target }
		}
		else = {
			planet = { save_event_target_as = ethic_shift_message_target }
		}
		if = {
			limit = {
				NOT = { has_carrier_flag = very_high_ethic_$ETHIC$_notification_sent@owner }
				colony = {
					pop_ethic_percentage = {
						percentage >= 0.75
						ethic = ethic_$ETHIC$
					}
				}
			}
			owner = {
				create_message = {
					type = MESSAGE_ETHIC_RISE_$ETHIC$
					localization =  MESSAGE_ETHIC_RISE_HIGHER_$ETHIC$_DESC
					target = event_target:ethic_shift_message_target
					days = 30
					variable = {
						type = name
						localization = PLANET
						scope = prev
					}
				}
			}
			set_timed_carrier_flag = {
				flag = very_high_ethic_$ETHIC$_notification_sent@owner
				years = 10 # you'll get a reminder after 10years if colony is still at the same state
			}
		}
		else_if = {
			limit = {
				NOR = {
					has_carrier_flag = high_ethic_$ETHIC$_notification_sent@owner
					has_carrier_flag = very_high_ethic_$ETHIC$_notification_sent@owner
				}
				colony = {
					pop_ethic_percentage = {
						percentage >= 0.1
						ethic = ethic_$ETHIC$
					}
				}
			}
			owner = {
				create_message = {
					type = MESSAGE_ETHIC_RISE_$ETHIC$
					localization =  MESSAGE_ETHIC_RISE_$ETHIC$_DESC
					target = event_target:ethic_shift_message_target
					days = 30
					variable = {
						type = name
						localization = PLANET
						scope = prev
					}
				}
			}
			set_timed_carrier_flag = {
				flag = high_ethic_$ETHIC$_notification_sent@owner
				years = 10 # you'll get a reminder after 10years if colony is still at the same state
			}
		}
	}
}
