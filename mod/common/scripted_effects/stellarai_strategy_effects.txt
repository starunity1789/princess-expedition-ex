# Country ROOT, managed AI. Monthly strategic facts read cheaply by budgets, plans and the market.
# The native AI decides wars and alliances; these facts only steer spending.
stellarai_refresh_strategic_facts = {
  if = {
    limit = {
      stellarai_verified_scientific_lead = yes
    }
    set_timed_country_flag = { flag = stellarai_science_lead days = 60 }
  }
  else = {
    remove_country_flag = stellarai_science_lead
  }
  # Preparation follows the AI's own war plan: a hostile or vassal-target attitude, or a recent claim.
  if = {
    limit = {
      is_at_war = no
      stellarai_conquest_identity = yes
      any_relation = {
        stellarai_war_target_candidate = { OBSERVER = root }
      }
    }
    set_timed_country_flag = { flag = stellarai_war_target days = 45 }
  }
  else = {
    remove_country_flag = stellarai_war_target
  }
  # Nobody expands undefended next to an empire that only wants it gone (alloys budget).
  if = {
    limit = {
      stellarai_expansion_phase = yes
      any_neighbor_country = {
        OR = {
          has_valid_civic = civic_fanatic_purifiers
          has_valid_civic = civic_hive_devouring_swarm
          has_valid_civic = civic_machine_terminator
        }
      }
    }
    set_timed_country_flag = { flag = stellarai_militant_neighbor days = 45 }
  }
  else = {
    remove_country_flag = stellarai_militant_neighbor
  }
  stellarai_refresh_army_need = yes
}

# Country ROOT. Armies are wanted while preparing for war or when threatened; war itself is checked live.
stellarai_refresh_army_need = {
  if = {
    limit = {
      OR = {
        has_country_flag = stellarai_war_target
        highest_threat > 30
        stellarai_hostile_neighbor_threat = yes
      }
    }
    set_timed_country_flag = { flag = stellarai_army_need days = 45 }
  }
  else = {
    remove_country_flag = stellarai_army_need
  }
}

# Country ROOT is the claimant and FROM the owner of the claimed system.
# Claims are a native war-preparation step, so preparation starts at once.
stellarai_note_claim_target = {
  if = {
    limit = {
      exists = from
      from = {
        is_country_type = default
      }
    }
    set_timed_relation_flag = { who = from flag = stellarai_claim_target days = 900 }
    if = {
      limit = {
        from = {
          stellarai_war_target_candidate = { OBSERVER = root }
        }
      }
      set_timed_country_flag = { flag = stellarai_war_target days = 45 }
      stellarai_refresh_army_need = yes
    }
  }
}
