# 皇女的远征军EX 3.0：AI 自动星球
# 写法借鉴《星球大战：夸特的黎明》的自动星球（产出 = 建筑系数 × 星球变量），但不改星球类型、
# 不用专用区划集：每颗星只留一级城市区划，政府分区里放 6 座自动建筑。
# 产能变量 atb_ai_auto_jobs = A（虚拟岗位数），见 common/scripted_variables/atb_ai_auto_variables.txt

# 星球作用域：把实际人口调整到 $SIZE$（每次增减 100）
atb_ai_auto_set_pops = {
	while = {
		count = 600
		limit = { pop_amount > $SIZE$ }
		random_owned_pop_group = {
			limit = { pop_group_size > 0 }
			kill_pop_group = { pop_group = this amount = 100 }
		}
	}
	while = {
		count = 20
		limit = { pop_amount < $SIZE$ }
		create_pop_group = { species = owner_main_species size = 100 }
	}
}

# 星球作用域：在 $DISTRICT$ 的政府分区里放下 6 座自动建筑
atb_ai_auto_place_buildings = {
	add_building = { district = $DISTRICT$ zone = zone_default building = building_atb_ai_core }
	add_building = { district = $DISTRICT$ zone = zone_default building = building_atb_ai_foundry }
	add_building = { district = $DISTRICT$ zone = zone_default building = building_atb_ai_power }
	add_building = { district = $DISTRICT$ zone = zone_default building = building_atb_ai_admin }
	add_building = { district = $DISTRICT$ zone = zone_default building = building_atb_ai_industry }
	add_building = { district = $DISTRICT$ zone = zone_default building = building_atb_ai_supply }
}

# 星球作用域：清空全部区划，只留一级城市区划（按区划集选对应的城市区划）
# 已知区划集之外的星球（宇宙创生世界等）不动区划，建筑直接放进原有的政府分区
atb_ai_auto_reset_districts = {
	if = {
		limit = { uses_district_set = habitat }
		remove_all_districts = yes
		if = { limit = { free_district_slots > 0 } add_district = district_hab_housing }
	}
	else_if = {
		limit = { uses_district_set = ring_world }
		remove_all_districts = yes
		if = {
			limit = { owner = { is_hive_empire = yes } }
			if = { limit = { free_district_slots > 0 } add_district = district_rw_hive }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			if = { limit = { free_district_slots > 0 } add_district = district_rw_nexus }
		}
		else = {
			if = { limit = { free_district_slots > 0 } add_district = district_rw_city }
		}
	}
	else_if = {
		limit = { uses_district_set = city_world }
		remove_all_districts = yes
		if = { limit = { free_district_slots > 0 } add_district = district_arcology_housing }
	}
	else_if = {
		limit = {
			OR = {
				uses_district_set = standard
				uses_district_set = volcanic_world
				uses_district_set = shattered_ring_world
				uses_district_set = hive_world
				uses_district_set = machine_world
			}
		}
		remove_all_districts = yes
		if = {
			limit = { owner = { is_hive_empire = yes } }
			if = { limit = { free_district_slots > 0 } add_district = district_hive }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			if = { limit = { free_district_slots > 0 } add_district = district_nexus }
		}
		else = {
			if = { limit = { free_district_slots > 0 } add_district = district_city }
		}
	}
}

# 星球作用域：放自动建筑，区划与上面的城市区划对应
atb_ai_auto_build = {
	if = {
		limit = { uses_district_set = habitat }
		atb_ai_auto_place_buildings = { DISTRICT = district_hab_housing }
	}
	else_if = {
		limit = { uses_district_set = ring_world }
		if = {
			limit = { owner = { is_hive_empire = yes } }
			atb_ai_auto_place_buildings = { DISTRICT = district_rw_hive }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			atb_ai_auto_place_buildings = { DISTRICT = district_rw_nexus }
		}
		else = { atb_ai_auto_place_buildings = { DISTRICT = district_rw_city } }
	}
	else_if = {
		limit = { uses_district_set = city_world }
		atb_ai_auto_place_buildings = { DISTRICT = district_arcology_housing }
	}
	else_if = {
		limit = {
			OR = {
				uses_district_set = standard
				uses_district_set = volcanic_world
				uses_district_set = shattered_ring_world
				uses_district_set = hive_world
				uses_district_set = machine_world
			}
		}
		if = {
			limit = { owner = { is_hive_empire = yes } }
			atb_ai_auto_place_buildings = { DISTRICT = district_hive }
		}
		else_if = {
			limit = { owner = { is_machine_empire = yes } }
			atb_ai_auto_place_buildings = { DISTRICT = district_nexus }
		}
		else = { atb_ai_auto_place_buildings = { DISTRICT = district_city } }
	}
	else = {
		add_building = building_atb_ai_core
		add_building = building_atb_ai_foundry
		add_building = building_atb_ai_power
		add_building = building_atb_ai_admin
		add_building = building_atb_ai_industry
		add_building = building_atb_ai_supply
	}
}

# 星球作用域：人口冻结（自然增长、组装都停），原版换主人时会清掉 planet_population_control，所以每次易手都重挂；
# 必须写 days = -1（同原版决议），不写的话修正挂不上（3.0 首个存档实测：57 颗只挂上 1 颗）
atb_ai_auto_freeze_pops = {
	if = {
		limit = { NOT = { has_modifier = planet_population_control } }
		add_modifier = { modifier = planet_population_control days = -1 }
	}
	if = {
		limit = { NOT = { has_modifier = planet_population_control_gestalt } }
		add_modifier = { modifier = planet_population_control_gestalt days = -1 }
	}
	if = {
		limit = { NOT = { has_modifier = planet_robot_assembly_control } }
		add_modifier = { modifier = planet_robot_assembly_control days = -1 }
	}
}

# 星球作用域：把普通殖民地转成自动星球
# 产能：AI 之间易手时保留原值；否则取「人口 ÷ 100」，但不低于新殖民地起点
atb_ai_auto_convert = {
	optimize_memory
	set_carrier_flag = atb_ai_auto
	if = {
		limit = { NOT = { is_variable_set = atb_ai_auto_jobs } }
		set_variable = { which = atb_ai_auto_jobs value = value:atb_ai_auto_pop_jobs }
		if = {
			limit = { owner = { atb_ai_auto_great_power = yes } }
			if = {
				limit = { check_variable = { which = atb_ai_auto_jobs value < @atb_ai_auto_start_gp } }
				set_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_start_gp }
			}
		}
		else_if = {
			limit = { check_variable = { which = atb_ai_auto_jobs value < @atb_ai_auto_start } }
			set_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_start }
		}
	}
	remove_all_buildings = yes
	atb_ai_auto_reset_districts = yes
	atb_ai_auto_build = yes
	atb_ai_auto_set_pops = { SIZE = @atb_ai_auto_pops }
	atb_ai_auto_freeze_pops = yes
}

# 星球作用域：AI 之间易手（产能保留）后、以及每年自检时，补回被原版清掉的修正，缺自治中枢时重建
# 3.4：每年自检也走这里——机械帝国的首府建筑会把自治中枢顶掉（系列 23 实测 3 颗），在这里拆掉重建；
# 星球变成不支持的区划集（宇宙创生世界等）时放回普通星球
atb_ai_auto_refresh = {
	if = {
		limit = { atb_ai_auto_supported_planet = no }
		atb_ai_auto_release = yes
	}
	else = {
		atb_ai_auto_freeze_pops = yes
		if = {
			limit = { NOT = { has_building = building_atb_ai_core } }
			remove_all_buildings = yes
			atb_ai_auto_reset_districts = yes
			atb_ai_auto_build = yes
		}
	}
}

# 星球作用域：放回普通 AI 星球（3.4，用于区划集变了、放不下自动建筑的星球）：
# 去掉标记、产能变量与人口冻结；自动建筑的 destroy_trigger 会自己拆掉，区划与原版建筑不动
atb_ai_auto_release = {
	remove_carrier_flag = atb_ai_auto
	clear_variable = atb_ai_auto_jobs
	remove_modifier = planet_population_control
	remove_modifier = planet_population_control_gestalt
	remove_modifier = planet_robot_assembly_control
}

# 星球作用域：玩家（或其他非自动国家）拿到自动星球——还原成刚殖民的新殖民地：
# 拆掉全部建筑，区划只留一级城市区划，人口删到 100，补上殖民地首府
atb_ai_auto_revert = {
	optimize_memory
	remove_carrier_flag = atb_ai_auto
	clear_variable = atb_ai_auto_jobs
	remove_modifier = planet_population_control
	remove_modifier = planet_population_control_gestalt
	remove_modifier = planet_robot_assembly_control
	remove_all_buildings = yes
	atb_ai_auto_reset_districts = yes
	atb_ai_auto_set_pops = { SIZE = @atb_ai_auto_reverted_pops }
	if = {
		limit = { uses_habitat_capitals = yes }
		add_building = building_hab_capital
	}
	else = {
		add_building = building_colony_shelter
	}
}

# 星球作用域：每年增长，封顶
# 3.20：大君主同盟领地不设上限；开局 100 年内与其他 AI 一样 +3/年，之后每年只 +1
atb_ai_auto_grow = {
	if = {
		limit = { owner = { has_country_flag = atbgp_power_3 } }
		if = {
			limit = { years_passed > 100 }
			change_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_growth_overlord_late }
		}
		else = {
			change_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_growth_gp }
		}
	}
	else_if = {
		limit = { owner = { atb_ai_auto_great_power = yes } }
		change_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_growth_gp }
		if = {
			limit = { check_variable = { which = atb_ai_auto_jobs value > @atb_ai_auto_cap_gp } }
			set_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_cap_gp }
		}
	}
	else = {
		change_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_growth }
		if = {
			limit = { check_variable = { which = atb_ai_auto_jobs value > @atb_ai_auto_cap } }
			set_variable = { which = atb_ai_auto_jobs value = @atb_ai_auto_cap }
		}
	}
}

# 国家作用域：AI 外交权重 ×3；换掉让 AI 不殖民的两个理念
atb_ai_auto_country_setup = {
	if = {
		limit = { NOT = { has_modifier = atb_ai_diplomacy } }
		add_modifier = { modifier = atb_ai_diplomacy }
	}
	# 3.0.2：原版 desired_colonizers 对「民营勘探者」「企业统治」恒为 0，AI 只能靠贸易值造赞助殖民船，
	# 而这条路 AI 实际不走（远星同盟整局只有母星；系列 7 另一个民营勘探者 AI 到 2295 年只有 2 颗星）。
	# 只对 AI 换：远星同盟换成群众福音（它唯心），其它巨企换成自由贸易者，寡头的企业统治换成采矿行会。
	if = {
		limit = { has_civic = civic_private_prospectors }
		force_remove_civic = civic_private_prospectors
		if = {
			limit = {
				has_country_flag = atbgp_power_2
				NOT = { has_civic = civic_gospel_of_the_masses }
			}
			force_add_civic = civic_gospel_of_the_masses
		}
		else_if = {
			limit = { NOT = { has_civic = civic_free_traders } }
			force_add_civic = civic_free_traders
		}
	}
	if = {
		limit = { has_civic = civic_corporate_dominion }
		force_remove_civic = civic_corporate_dominion
		if = {
			limit = { NOT = { has_civic = civic_mining_guilds } }
			force_add_civic = civic_mining_guilds
		}
	}
}
