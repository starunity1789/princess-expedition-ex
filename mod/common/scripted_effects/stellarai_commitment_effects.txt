# Planet scope. Full local refresh for colony lifecycle events; never chooses national fields.
stellarai_refresh_planet_state = {
  stellarai_classify_planet_structure = yes
  stellarai_refresh_planet_eligibility = yes
}

# Planet scope. Structural commitments from completed development; no transactions or scheduling.
# Zone families include every district-set variant (hive, nexus, ring, arcology, wilderness, habitat).
stellarai_classify_planet_structure = {
  if = {
    limit = {
      NOR = {
        has_building = building_bureaucratic_1
        has_building = building_bureaucratic_2
        has_building = building_bureaucratic_3
        has_building = building_fe_administration_1
        has_building = building_fe_administration_2
        has_building = building_league_offices
        stellarai_has_unity_zone = yes
        num_districts = { type = district_arcology_administrative value > 0 }
      }
    }
    set_planet_flag = stellarai_invalid_for_unity
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_unity
  }
  # Research presence, then established research development.
  if = {
    limit = {
      OR = {
        is_capital = yes
        has_any_research_zone = yes
        num_districts = { type = district_hab_science value > 0 }
        num_districts = { type = district_rw_science value > 0 }
        has_building = building_research_lab_1
        has_building = building_research_lab_2
        has_building = building_research_lab_3
        has_building = building_physics_lab_1
        has_building = building_physics_lab_2
        has_building = building_physics_lab_3
        has_building = building_engineering_facility_1
        has_building = building_engineering_facility_2
        has_building = building_engineering_facility_3
        has_building = building_biolab_1
        has_building = building_biolab_2
        has_building = building_biolab_3
      }
    }
    set_planet_flag = stellarai_research_commitment_soft
  }
  else = {
    remove_planet_flag = stellarai_research_commitment_soft
  }
  if = {
    limit = {
      OR = {
        calc_true_if = {
          amount >= 2
          stellarai_has_generic_research_zone = yes
          stellarai_has_research_unity_zone = yes
          has_any_research_physics_zone = yes
          has_any_research_society_zone = yes
          has_any_research_engineering_zone = yes
          num_districts = { type = district_hab_science value > 0 }
          num_districts = { type = district_rw_science value > 0 }
        }
        stellarai_has_research_zone_pair = yes
        num_districts = { type = district_hab_science value > 1 }
        num_districts = { type = district_rw_science value > 1 }
        num_buildings = { type = building_research_lab_1 value > 2 }
        num_buildings = { type = building_research_lab_2 value > 1 }
        num_buildings = { type = building_research_lab_3 value > 0 }
        num_buildings = { type = building_physics_lab_1 value > 1 }
        num_buildings = { type = building_physics_lab_2 value > 0 }
        num_buildings = { type = building_physics_lab_3 value > 0 }
        num_buildings = { type = building_engineering_facility_1 value > 1 }
        num_buildings = { type = building_engineering_facility_2 value > 0 }
        num_buildings = { type = building_engineering_facility_3 value > 0 }
        num_buildings = { type = building_biolab_1 value > 1 }
        num_buildings = { type = building_biolab_2 value > 0 }
        num_buildings = { type = building_biolab_3 value > 0 }
      }
    }
    set_planet_flag = stellarai_research_commitment_hard
  }
  else = {
    remove_planet_flag = stellarai_research_commitment_hard
  }
  if = {
    limit = {
      NOT = { has_planet_flag = stellarai_research_commitment_soft }
      NOR = {
        has_building = building_archaeostudies_faculty
        has_building = building_vultaum_reality_computer
        has_building = building_institute
        has_building = building_shroud_observatory_1
        has_building = building_shroud_observatory_2
        has_building = building_shroud_observatory_3
        has_building = building_ranger_lodge
        check_modifier_value = { modifier = planet_researchers_produces_mult value > 0.09 }
        check_modifier_value = { modifier = planet_researchers_physics_research_produces_mult value > 0.09 }
        check_modifier_value = { modifier = planet_researchers_society_research_produces_mult value > 0.09 }
        check_modifier_value = { modifier = planet_researchers_engineering_research_produces_mult value > 0.09 }
      }
    }
    set_planet_flag = stellarai_invalid_for_research
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_research
  }
  # Industry presence, then established industrial development.
  if = {
    limit = {
      OR = {
        has_any_industrial_zone = yes
        has_building = building_factory_1
        has_building = building_factory_2
        has_building = building_factory_3
        has_building = building_foundry_1
        has_building = building_foundry_2
        has_building = building_foundry_3
      }
    }
    set_planet_flag = stellarai_industry_commitment_soft
  }
  else = {
    remove_planet_flag = stellarai_industry_commitment_soft
  }
  if = {
    limit = {
      OR = {
        has_mixed_industrial_zone = yes
        AND = {
          has_any_factory_zone = yes
          has_any_foundry_zone = yes
        }
        num_buildings = { type = building_factory_2 value > 0 }
        num_buildings = { type = building_factory_3 value > 0 }
        num_buildings = { type = building_foundry_2 value > 0 }
        num_buildings = { type = building_foundry_3 value > 0 }
        num_buildings = { type = building_factory_1 value > 1 }
        num_buildings = { type = building_foundry_1 value > 1 }
        AND = {
          has_building = building_factory_1
          has_building = building_foundry_1
        }
      }
    }
    set_planet_flag = stellarai_industry_commitment_hard
  }
  else = {
    remove_planet_flag = stellarai_industry_commitment_hard
  }
  if = {
    limit = {
      NOT = { has_planet_flag = stellarai_industry_commitment_soft }
      OR = {
        check_modifier_value = { modifier = planet_jobs_produces_mult value < -0.09 }
        AND = {
          check_modifier_value = { modifier = planet_metallurgists_produces_mult value < -0.09 }
          check_modifier_value = { modifier = planet_artisans_produces_mult value < -0.09 }
        }
      }
    }
    set_planet_flag = stellarai_invalid_for_industry
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_industry
  }
  # Rural roles. Established research or industry excludes them for AI designations.
  if = {
    limit = {
      OR = {
        has_planet_flag = stellarai_research_commitment_hard
        has_planet_flag = stellarai_industry_commitment_hard
        AND = {
          num_districts = { type = district_mining value = 0 }
          num_districts = { type = district_melting value = 0 }
          has_any_normal_mining_zone = no
          OR = {
            num_districts = { type = district_generator value > 0 }
            has_any_normal_generator_zone = yes
            num_districts = { type = district_farming value > 0 }
            has_any_normal_agriculture_zone = yes
          }
          NOR = {
            has_modifier = asteroid_belt
            has_modifier = mineral_rich
            has_modifier = ultra_rich
            has_modifier = extensive_moon_system
            has_modifier = asteroid_impacts
            has_modifier = carbon_world
            has_modifier = volcanic_mineral_rich
            has_modifier = volcanic_ultra_rich
            has_modifier = volcanic_subterranean
            has_modifier = volcanic_asteroid_impacts
            has_modifier = volcanic_hollow_planet
            has_modifier = volcanic_lavafall
          }
        }
      }
    }
    set_planet_flag = stellarai_invalid_for_mining
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_mining
  }
  if = {
    limit = {
      OR = {
        has_planet_flag = stellarai_research_commitment_hard
        has_planet_flag = stellarai_industry_commitment_hard
        AND = {
          num_districts = { type = district_farming value = 0 }
          has_any_normal_agriculture_zone = no
          OR = {
            num_districts = { type = district_generator value > 0 }
            has_any_normal_generator_zone = yes
            num_districts = { type = district_mining value > 0 }
            num_districts = { type = district_melting value > 0 }
            has_any_normal_mining_zone = yes
          }
          NOT = {
            check_modifier_value = { modifier = pop_farmer_bonus_workforce_mult value > 0.09 }
          }
        }
      }
    }
    set_planet_flag = stellarai_invalid_for_farming
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_farming
  }
  if = {
    limit = {
      OR = {
        has_planet_flag = stellarai_research_commitment_hard
        has_planet_flag = stellarai_industry_commitment_hard
        AND = {
          num_districts = { type = district_generator value = 0 }
          num_districts = { type = district_geothermal value = 0 }
          has_any_normal_generator_zone = no
          OR = {
            num_districts = { type = district_farming value > 0 }
            has_any_normal_agriculture_zone = yes
            num_districts = { type = district_mining value > 0 }
            num_districts = { type = district_melting value > 0 }
            has_any_normal_mining_zone = yes
          }
          NOR = {
            has_modifier = hazardous_weather
            has_modifier = wild_storms
            has_modifier = abundant_geothermal_activity
          }
        }
      }
    }
    set_planet_flag = stellarai_invalid_for_generator
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_generator
  }
  if = {
    limit = {
      NOR = {
        has_building = building_stronghold
        has_building = building_fortress
        has_building = building_order_keep
        has_building = building_order_castle
        has_any_fortress_zone = yes
      }
    }
    set_planet_flag = stellarai_invalid_for_military
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_military
  }
  if = {
    limit = {
      OR = {
        has_any_normal_generator_zone = yes
        has_any_normal_mining_zone = yes
        has_any_normal_agriculture_zone = yes
        num_districts = { type = district_generator value > 0 }
        num_districts = { type = district_farming value > 0 }
        num_districts = { type = district_mining value > 0 }
      }
    }
    set_planet_flag = stellarai_rural_commitment_soft
  }
  else = {
    remove_planet_flag = stellarai_rural_commitment_soft
  }
  if = {
    limit = {
      OR = {
        has_building = building_commercial_zone
        has_building = building_commercial_megaplex
        has_building = building_galactic_stock_exchange
        has_building = building_maintenance_depot
        has_any_trade_zone = yes
      }
    }
    set_planet_flag = stellarai_trade_commitment_soft
  }
  else = {
    remove_planet_flag = stellarai_trade_commitment_soft
  }
}

# Planet scope. The research construction lock follows claims and the live economy.
stellarai_refresh_planet_eligibility = {
  if = {
    limit = {
      OR = {
        check_modifier_value = { modifier = planet_researchers_produces_mult value < -0.09 }
        check_modifier_value = { modifier = planet_researchers_physics_research_produces_mult value < -0.09 }
        check_modifier_value = { modifier = planet_researchers_society_research_produces_mult value < -0.09 }
        check_modifier_value = { modifier = planet_researchers_engineering_research_produces_mult value < -0.09 }
        AND = {
          is_capital = no
          NOT = {
            has_planet_flag = stellarai_research_plan_claimed
          }
        }
        owner = {
          stellarai_planning_economic_crisis = yes
        }
        AND = {
          is_capital = no
          owner = {
            NOT = {
              stellarai_planning_foundation_ready = yes
            }
          }
        }
      }
    }
    set_planet_flag = stellarai_invalid_for_research_to_build
  }
  else = {
    remove_planet_flag = stellarai_invalid_for_research_to_build
  }
}

# Planet scope. A colony leaving StellarAI management keeps no stale designation flags.

  remove_planet_flag = stellarai_invalid_for_mining
  remove_planet_flag = stellarai_invalid_for_farming
  remove_planet_flag = stellarai_invalid_for_generator
  remove_planet_flag = stellarai_invalid_for_industry
  remove_planet_flag = stellarai_invalid_for_research
  remove_planet_flag = stellarai_invalid_for_military
  remove_planet_flag = stellarai_invalid_for_research_to_build
  remove_planet_flag = stellarai_rural_commitment_soft
  remove_planet_flag = stellarai_trade_commitment_soft
  remove_planet_flag = stellarai_industry_commitment_soft
  remove_planet_flag = stellarai_industry_commitment_hard
  remove_planet_flag = stellarai_research_commitment_soft
  remove_planet_flag = stellarai_research_commitment_hard
  remove_planet_flag = stellarai_research_plan_claimed
  remove_planet_flag = stellarai_capital_specialist_physics
  remove_planet_flag = stellarai_capital_specialist_engineering
  remove_planet_flag = stellarai_capital_specialist_society
}

# Country ROOT. Startup/new-AI bootstrap uses the regular research pass once.
# It reconciles existing colonies immediately and never replays monthly transactions.
stellarai_initialize_planning = {
  if = {
    limit = {
      NOT = {
        has_country_flag = stellarai_planning_initialized
      }
    }
    set_country_flag = stellarai_annual_refresh_due
    set_country_flag = stellarai_schema_2
    stellarai_plan_research_worlds = yes
    set_country_flag = stellarai_planning_initialized
  }
  # Existing campaigns: reclassify every colony once with the current rules.
  if = {
    limit = {
      NOT = {
        has_country_flag = stellarai_schema_2
      }
    }
    set_country_flag = stellarai_schema_2
    set_country_flag = stellarai_annual_refresh_due
    remove_country_flag = stellarai_assessment_current
    clear_variable = stellarai_coalition_confident
    clear_variable = stellarai_coalition_ratio
    clear_variable = stellarai_assessment_due_day
  }
}

# Country scope; ROOT must be this country. Synchronous monthly planning.
stellarai_plan_research_worlds = {
  if = {
    limit = {
      is_ai = yes
      is_nomadic = no
    }
    stellarai_refresh_capital_science = yes
    # Snapshot only immutable facts for this synchronous, non-transactional pass.
    set_country_flag = stellarai_planning_context
    remove_country_flag = stellarai_planning_crisis
    remove_country_flag = stellarai_planning_foundation
    if = {
      limit = {
        stellarai_economic_crisis = yes
      }
      set_country_flag = stellarai_planning_crisis
    }
    if = {
      limit = {
        stellarai_foundation_ready = yes
      }
      set_country_flag = stellarai_planning_foundation
    }
    # One monthly scan replaces per-planet scans in colony designation potentials.
    if = {
      limit = {
        stellarai_has_consumer_goods_world = yes
      }
      set_country_flag = stellarai_has_cg_world
    }
    else = {
      remove_country_flag = stellarai_has_cg_world
    }
    # Reconcile designations and count surviving claims in one traversal.
    # This scratch counter exists only for this synchronous country update.
    set_variable = { which = stellarai_research_claim_count value = 0 }
    every_owned_planet = {
      planet = {
        if = {
          limit = {
            is_colony = yes
            owner = {
              has_country_flag = stellarai_annual_refresh_due
            }
          }
          stellarai_classify_planet_structure = yes
        }
        if = {
          limit = {
            is_colony = yes
            is_capital = no
          }
          if = {
            limit = {
              has_designation = col_research
            }
            if = {
              limit = {
                stellarai_research_plan_suitable = yes
              }
              set_planet_flag = stellarai_research_plan_claimed
            }
            else = {
              remove_planet_flag = stellarai_research_plan_claimed
            }
          }
          else_if = {
            limit = {
              OR = {
                has_designation = col_habitat_research
                has_designation = col_ring_research
              }
            }
            set_planet_flag = stellarai_research_plan_claimed
          }
          # Established science development claims its own research role.
          else_if = {
            limit = {
              has_planet_flag = stellarai_research_commitment_hard
              NOT = { has_planet_flag = stellarai_industry_commitment_hard }
              NOT = { has_planet_flag = stellarai_research_plan_claimed }
              stellarai_research_plan_suitable = yes
            }
            set_planet_flag = stellarai_research_plan_claimed
          }
        }
        # Count all claims as before, including reserved and special research worlds.
        if = {
          limit = {
            has_planet_flag = stellarai_research_plan_claimed
          }
          owner = {
            change_variable = { which = stellarai_research_claim_count value = 1 }
          }
        }
        # Refresh existing planets in this traversal. Candidate selection does
        # not read these eligibility flags; newly claimed planets refresh below.
        if = {
          limit = {
            is_colony = yes
          }
          stellarai_refresh_planet_eligibility = yes
        }
      }
    }
    # At two colonies, claim only a genuine research candidate. This keeps
    # the capital's early science start while preserving the sole developing
    # industrial or rural colony. At three colonies, a fallback still needs
    # research evidence or substantial uncommitted urban capacity.
    # The iterator is colony scope, so planet-only tests enter planet first.
    if = {
      limit = {
        num_owned_planets > 1
        check_variable = { which = stellarai_research_claim_count value < 1 }
      }
      random_owned_planet = {
        limit = {
          planet = {
            OR = {
              stellarai_research_early_candidate = yes
              AND = {
                owner = {
                  num_owned_planets > 2
                }
                stellarai_research_fallback_candidate = yes
              }
            }
          }
        }
        weights = {
          base = 10
          modifier = {
            factor = 20
            planet = {
              stellarai_good_research_candidate = yes
            }
          }
          # A poor research modifier is a fallback cost, not a reason to
          # leave an otherwise suitable empire without any research colony.
          modifier = {
            factor = 0.10
            planet = {
              OR = {
                check_modifier_value = { modifier = planet_researchers_produces_mult value < -0.09 }
                check_modifier_value = { modifier = planet_researchers_physics_research_produces_mult value < -0.09 }
                check_modifier_value = { modifier = planet_researchers_society_research_produces_mult value < -0.09 }
                check_modifier_value = { modifier = planet_researchers_engineering_research_produces_mult value < -0.09 }
              }
            }
          }
          modifier = {
            factor = 0.25
            planet = {
              has_planet_flag = stellarai_rural_commitment_soft
            }
          }
          modifier = {
            factor = 0.10
            planet = {
              has_planet_flag = stellarai_industry_commitment_soft
            }
          }
        }
        planet = {
          set_planet_flag = stellarai_research_plan_claimed
          owner = {
            change_variable = { which = stellarai_research_claim_count value = 1 }
          }
          stellarai_refresh_planet_eligibility = yes
        }
      }
    }
    # From five colonies, reserve a second good candidate.
    if = {
      limit = {
        num_owned_planets > 4
        check_variable = { which = stellarai_research_claim_count value < 2 }
      }
      random_owned_planet = {
        limit = {
          planet = {
            stellarai_research_claim_candidate = yes
            stellarai_good_research_candidate = yes
          }
        }
        weights = {
          base = 10
          modifier = {
            factor = 0.25
            planet = {
              has_planet_flag = stellarai_rural_commitment_soft
            }
          }
          modifier = {
            factor = 0.10
            planet = {
              has_planet_flag = stellarai_industry_commitment_soft
            }
          }
        }
        planet = {
          set_planet_flag = stellarai_research_plan_claimed
          owner = {
            change_variable = { which = stellarai_research_claim_count value = 1 }
          }
          stellarai_refresh_planet_eligibility = yes
        }
      }
    }
    # With a working economy, >5/9/13/17/21 colonies target 2/3/4/5/6 research claims.
    remove_country_flag = stellarai_good_candidates_exhausted
    inline_script = { script = stellarai/research_claim_good_slot PLANETS = 5 COUNT = 2 }
    inline_script = { script = stellarai/research_claim_good_slot PLANETS = 9 COUNT = 3 }
    inline_script = { script = stellarai/research_claim_good_slot PLANETS = 13 COUNT = 4 }
    inline_script = { script = stellarai/research_claim_good_slot PLANETS = 17 COUNT = 5 }
    inline_script = { script = stellarai/research_claim_good_slot PLANETS = 21 COUNT = 6 }
    remove_country_flag = stellarai_good_candidates_exhausted
    clear_variable = stellarai_research_claim_count
    remove_country_flag = stellarai_annual_refresh_due
    remove_country_flag = stellarai_planning_context
    remove_country_flag = stellarai_planning_crisis
    remove_country_flag = stellarai_planning_foundation
  }
}

# Country scope; ROOT must be this country. Used on initialization/capital change and monthly.
stellarai_refresh_capital_science = {
  # Monthly capital science selection: actual research output decides the
  # two distinct specialist fields; physics and then engineering win ties.
  export_resource_income_to_variable = { resource = physics_research variable = stellarai_physics_income }
  export_resource_income_to_variable = { resource = society_research variable = stellarai_society_income }
  export_resource_income_to_variable = { resource = engineering_research variable = stellarai_engineering_income }
  # Some zero-income exports leave no variable. Normalize before comparisons.
  if = {
    limit = {
      NOT = {
        is_variable_set = stellarai_physics_income
      }
    }
    set_variable = { which = stellarai_physics_income value = 0 }
  }
  if = {
    limit = {
      NOT = {
        is_variable_set = stellarai_society_income
      }
    }
    set_variable = { which = stellarai_society_income value = 0 }
  }
  if = {
    limit = {
      NOT = {
        is_variable_set = stellarai_engineering_income
      }
    }
    set_variable = { which = stellarai_engineering_income value = 0 }
  }
  if = {
    limit = {
      exists = capital_scope
      capital_scope = {
        owner = {
          is_same_value = root
        }
      }
    }
    capital_scope = {
      planet = {
        # The capital is a science center, not a claimed research colony.
        remove_planet_flag = stellarai_research_plan_claimed
        set_planet_flag = stellarai_research_commitment_soft
        remove_planet_flag = stellarai_capital_specialist_physics
        remove_planet_flag = stellarai_capital_specialist_engineering
        remove_planet_flag = stellarai_capital_specialist_society
        if = {
          limit = {
            root = {
              check_variable = { which = stellarai_physics_income value <= stellarai_society_income }
              check_variable = { which = stellarai_physics_income value <= stellarai_engineering_income }
            }
          }
          set_planet_flag = stellarai_capital_specialist_physics
          if = {
            limit = {
              root = {
                check_variable = { which = stellarai_engineering_income value <= stellarai_society_income }
              }
            }
            set_planet_flag = stellarai_capital_specialist_engineering
          }
          else = {
            set_planet_flag = stellarai_capital_specialist_society
          }
        }
        else_if = {
          limit = {
            root = {
              check_variable = { which = stellarai_engineering_income value <= stellarai_society_income }
            }
          }
          set_planet_flag = stellarai_capital_specialist_engineering
          if = {
            limit = {
              root = {
                check_variable = { which = stellarai_physics_income value <= stellarai_society_income }
              }
            }
            set_planet_flag = stellarai_capital_specialist_physics
          }
          else = {
            set_planet_flag = stellarai_capital_specialist_society
          }
        }
        else = {
          set_planet_flag = stellarai_capital_specialist_society
          if = {
            limit = {
              root = {
                check_variable = { which = stellarai_physics_income value <= stellarai_engineering_income }
              }
            }
            set_planet_flag = stellarai_capital_specialist_physics
          }
          else = {
            set_planet_flag = stellarai_capital_specialist_engineering
          }
        }
