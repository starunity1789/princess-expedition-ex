# Country scope. Assign once; country death never reshuffles existing buckets.
stellarai_assign_planning_bucket = {
  if = {
    limit = {
      NOT = {
        is_variable_set = stellarai_planning_bucket
      }
    }
    if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_1
      }
      remove_global_flag = stellarai_next_planning_bucket_1
      set_global_flag = stellarai_next_planning_bucket_2
      set_variable = { which = stellarai_planning_bucket value = 1 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_2
      }
      remove_global_flag = stellarai_next_planning_bucket_2
      set_global_flag = stellarai_next_planning_bucket_3
      set_variable = { which = stellarai_planning_bucket value = 2 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_3
      }
      remove_global_flag = stellarai_next_planning_bucket_3
      set_global_flag = stellarai_next_planning_bucket_4
      set_variable = { which = stellarai_planning_bucket value = 3 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_4
      }
      remove_global_flag = stellarai_next_planning_bucket_4
      set_global_flag = stellarai_next_planning_bucket_5
      set_variable = { which = stellarai_planning_bucket value = 4 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_5
      }
      remove_global_flag = stellarai_next_planning_bucket_5
      set_global_flag = stellarai_next_planning_bucket_6
      set_variable = { which = stellarai_planning_bucket value = 5 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_6
      }
      remove_global_flag = stellarai_next_planning_bucket_6
      set_global_flag = stellarai_next_planning_bucket_7
      set_variable = { which = stellarai_planning_bucket value = 6 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_7
      }
      remove_global_flag = stellarai_next_planning_bucket_7
      set_global_flag = stellarai_next_planning_bucket_8
      set_variable = { which = stellarai_planning_bucket value = 7 }
    }
    else_if = {
      limit = {
        has_global_flag = stellarai_next_planning_bucket_8
      }
      remove_global_flag = stellarai_next_planning_bucket_8
      # Wrap: absence of a cursor flag means bucket zero.
      set_variable = { which = stellarai_planning_bucket value = 8 }
    }
    else = {
      set_variable = { which = stellarai_planning_bucket value = 0 }
      set_global_flag = stellarai_next_planning_bucket_1
    }
  }
}

# Country ROOT. One fixed worker slot (2..26 days after the pulse) per AI per month.
# No random delay, recursive rescheduling, pending-flag timeout or replacement retry.
# Due dates are consumed before work; duplicate same-day delivery cannot run twice.
stellarai_queue_monthly_planning = {
  stellarai_assign_planning_bucket = yes
  if = {
    limit = {
      check_variable = { which = stellarai_planning_bucket value = 0 }
    }
    inline_script = { script = stellarai/queue_planning_slot RESEARCH_DELAY = 2 }
  }
  else_if = {
    limit = {
      check_variable = { which = stellarai_planning_bucket value = 1 }
