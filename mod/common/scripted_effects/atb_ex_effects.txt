# 国家作用域：本国还没有银河规则就加上（存储上限 +100K、非生物船基础舰体翻倍）
atb_ex_give_galaxy_rules = {
	if = {
		limit = { NOT = { has_modifier = atb_ex_galaxy_rules } }
		add_modifier = {
			modifier = atb_ex_galaxy_rules
			days = -1
		}
	}
	# 3.3：所有国家亚光速 ×1.2，不论启用了哪些 MOD（3.1～3.2 是启用 Ships in Scaling 时 ×12）。
	# 沿用同一个修正名，旧存档里已挂着的 ×12 读档即变成 ×1.2
	if = {
		limit = { NOT = { has_modifier = atb_ex_sublight_rs } }
		add_modifier = {
			modifier = atb_ex_sublight_rs
			days = -1
		}
	}
}

# 2.8：居住站科技开局就给所有帝国（tech_habitat_1/2/3）。
# 原版前置是 tech_starbase_3/4/5，give_technology 不会连带前置；建造居住站只看
# tech_habitat_1，tech_habitat_2/3 各给 habitat_max_districts_add = 1。
# 按原版 tech_habitat_* 的 potential 排除荒野与游牧帝国，并且只发给常规国家
# （陨落帝国、强盗、天选民之类不发）。已有 tech_habitat_3 就跳过。

# 国家作用域：本国符合条件就发
atb_ex_give_habitat_techs = {
	if = {
		limit = {
			is_country_type = default
			is_wilderness_empire = no
			is_nomadic = no
			NOT = { has_technology = tech_habitat_3 }
		}
		give_technology = { tech = tech_habitat_1 message = no }
		give_technology = { tech = tech_habitat_2 message = no }
		give_technology = { tech = tech_habitat_3 message = no }
	}
}

# 全体国家补发（开局一次、每年兜底一次）
atb_ex_grant_habitat_techs = {
	every_country = {
		limit = {
			is_country_type = default
			NOT = { has_technology = tech_habitat_3 }
		}
		atb_ex_give_habitat_techs = yes
	}
}

	capital_scope = { save_global_event_target_as = atb_rl_old_capital }
	capital_star = { save_global_event_target_as = atb_rl_old_system }

	# 1. 生成星团

	}
	else_if = {
		limit = { galaxy_size = medium }
		no_scope = {
			spawn_system = {
				min_distance >= 495
				max_distance <= 503
				min_orientation_angle = 88
				max_orientation_angle = 92
				initializer = atb_rl_home
				hyperlane = no
			}
		}
	}
	else_if = {
		limit = { galaxy_size = large }
		no_scope = {
			spawn_system = {
				min_distance >= 545
				max_distance <= 553
				min_orientation_angle = 88
				max_orientation_angle = 92
				initializer = atb_rl_home

		no_scope = {
			spawn_system = {
				min_distance >= 545
				max_distance <= 553
				min_orientation_angle = 88
				max_orientation_angle = 92
				initializer = atb_rl_home
				hyperlane = no
			}
		}

	else = {
		no_scope = {
			spawn_system = {
				min_distance >= 545
				max_distance <= 553
				min_orientation_angle = 88
				max_orientation_angle = 92
				initializer = atb_rl_home
				hyperlane = no
			}

			min_distance >= 55
			max_distance <= 62
			min_orientation_angle = 267
			max_orientation_angle = 273

			min_distance >= 32

			min_orientation_angle = 356
			max_orientation_angle = 4

			hyperlane = no

	set_spawn_system_batch = end

	event_target:atb_rl_sys_home = { add_hyperlane = { from = this to = event_target:atb_rl_sys_gateway } }
	event_target:atb_rl_sys_home = { add_hyperlane = { from = this to = event_target:atb_rl_sys_lgate } }
	event_target:atb_rl_sys_home = { add_hyperlane = { from = this to = event_target:atb_rl_sys_preftl } }
	event_target:atb_rl_sys_home = { add_hyperlane = { from = this to = event_target:atb_rl_sys_blackhole } }

	event_target:atb_rl_sys_lgate = { add_hyperlane = { from = this to = event_target:atb_rl_sys_blackhole } }

	event_target:atb_rl_new_capital = {
		set_owner = event_target:atb_rl_country
		set_controller = event_target:atb_rl_country
		remove_planet_flag = ignore_startup_effect

	event_target:atb_rl_old_capital = {

				create_pop_group = {
					pop_group = event_target:atb_rl_src_group
					size = 1
					effect = { save_event_target_as = atb_rl_dst_group }
				}
			}
			transfer_pop_amount = {
				source = event_target:atb_rl_src_group
				target = event_target:atb_rl_dst_group
				percentage = 1
			}
		}
	}
	event_target:atb_rl_new_capital = {
		set_capital = yes

		# （建筑区划里已经包含合成繁衍的废弃基因诊所、身份存储库和 2 个废弃城市，不要再手动加）

			generate_start_buildings_and_districts = yes
		}

		set_planet_size = 25
		set_planet_flag = atb_fortress_deposits_added

		add_deposit = d_atb_star_sea_monument
		add_deposit = d_atb_sensor_array

		add_deposit = d_atb_elemental_seal_institute
	}
	species = { set_species_homeworld = event_target:atb_rl_new_capital }

	# 3. 舰队随行
	every_owned_fleet = {
		limit = { NOT = { is_ship_class = shipclass_starbase } }
		set_location = event_target:atb_rl_sys_home
	}

	# 4. 旧母星：要塞已经开走，只剩一颗荒芜星球；拆掉旧星港，星系改名
	event_target:atb_rl_old_capital = {
		remove_planet_flag = atb_origin_fortress
		remove_all_buildings = yes
		destroy_colony = yes
		clear_deposits = yes
		change_pc = pc_barren
	}
	event_target:atb_rl_old_system = {
		set_name = "atb_system_old_origin"
		if = {
			limit = { exists = starbase }
			starbase = { fleet = { destroy_fleet = this } }
		}
	}

	# 5. 只有新母星系已勘测；星团其余星系保持未探索，由科研船自己去勘测
	event_target:atb_rl_sys_home = {
		set_surveyed = { surveyed = yes surveyor = event_target:atb_rl_country }
		every_system_planet = {
			set_surveyed = { surveyed = yes surveyor = event_target:atb_rl_country }
		}
	}
}

# 星球作用域：生成「无形链接」的前超光速土著，物种固定为龙族
# 肖像来自合并进本 MOD 的 Ancient Dragonkin Animated Portrait Pack（adpp_ 前缀），
# 该肖像组注册在爬行类(REP)与炼狱类(INF)下，所以物种类一并固定为 REP。
#
# 这里照搬原版 generate_pre_ftls_on_planet 的流程，只把 create_pre_ftl_species 换成
# 自己建物种：原版 create_pre_ftl_empire 取的是 last_created_species，所以能接上。
# （先生成再用 change_species_characteristics 改肖像的做法不行：那个效果的 portrait
#   只认具体肖像、不认肖像组，而且物种类仍是随机的，和龙族肖像对不上。）
atb_ex_generate_dragonkin_preftl = {
	select_pre_ftl_origin = yes
	# 3.6：种族名固定为「龙族」，特质固定为 Nivlac + Robust + Psionic + Stargazer（直接指定，不受点数限制）；
	# 气候偏好取母星类型（ideal_planet_class = this，同原版写法）
	create_species = {
		name = atb_species_rl_dragonkin
		plural = atb_species_rl_dragonkin
		adjective = atb_species_rl_dragonkin
		class = REP
		portrait = adpp_ancient_dragonkin
		homeworld = THIS
		traits = {
			ideal_planet_class = this
			trait = trait_nivlac
			trait = trait_robust
			trait = trait_psionic
			trait = trait_stargazer
		}
	}
	create_pre_ftl_empire = yes
	setup_pre_ftl_planet = yes
	pre_ftl_clean_up_effect = yes
}
