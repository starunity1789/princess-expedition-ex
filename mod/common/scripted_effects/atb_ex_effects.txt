# 3.10 无作用域：生成天选民，并把伊索姆之门的虫洞出口固定在大君主同盟领地（保底列强 3）的母星系。
# 原版 the_chosen_gate_initializer 用 random_system 随机挑出口，排除 empire_cluster 星系；
# 生成前给其它所有星系临时挂上 empire_cluster，出口就只能落在大君主母星，生成完立刻摘掉。
# 然后天选民改用大君主的主种族（两边的种族洁癖对彼此无效），并与大君主建立通讯：
# 原版首次接触（fircon.3515）与出门（fircon.3505）的宣战都以「还没有通讯」为前提，就不会再对大君主宣战；
# 开门（任一方第一次穿过伊索姆之门的虫洞）后由 atb_ex.8 让天选民与大君主结盟。
atb_ex_spawn_chosen_fixed = {
	random_country = {
		limit = { has_country_flag = atbgp_power_3 exists = capital_scope }
		save_global_event_target_as = atb_ex_overlord
		capital_scope.solar_system = { save_global_event_target_as = atb_ex_overlord_home }
		species = { save_global_event_target_as = atb_ex_overlord_species }
	}
	if = {
		limit = { exists = event_target:atb_ex_overlord_home }
		every_system = {
			limit = {
				NOT = { has_star_flag = empire_cluster }
				NOT = { is_same_value = event_target:atb_ex_overlord_home }
			}
			set_star_flag = atb_ex_tmp_cluster
			set_star_flag = empire_cluster
		}
	}
	atb_ex_spawn_chosen = yes
	every_system = {
		limit = { has_star_flag = atb_ex_tmp_cluster }
		remove_star_flag = empire_cluster
		remove_star_flag = atb_ex_tmp_cluster
	}
	if = {
		limit = {
			exists = event_target:chosen_empire
			exists = event_target:atb_ex_overlord
		}
		event_target:chosen_empire = {
			change_dominant_species = {
				species = event_target:atb_ex_overlord_species
				change_all = yes
			}
			establish_communications_no_message = event_target:atb_ex_overlord
		}
		event_target:atb_ex_overlord = {
			establish_communications_no_message = event_target:chosen_empire
		}
	}
}
