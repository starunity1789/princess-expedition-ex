can_embrace_faction = {
	custom_tooltip = {
		fail_text = NOT_CROWDSOURCING_EMBRACE
		owner = {
			OR = {
				is_$ETHIC$ = no
				is_crowdsourcing_empire = no
			}
		}
	}
	custom_tooltip = {
		fail_text = EMBRACE_FACTION_COOLDOWN
		owner = {
			NOT = { has_modifier = embraced_faction_timer }
		}
	}

	is_in_political_reorientation = no

	support > 0.20
	if = {
		limit = {
			is_faction_fanatic = yes
			owner = { is_fanatic_$ETHIC$ = yes }
		}
		owner = {
			custom_tooltip = {
				success_text = tooltip_not_abs_$ETHIC$
				fail_text = tooltip_not_abs_$ETHIC$
				NOT = { has_ethic = ethic_absolute_$ETHIC$ }
			}
			custom_tooltip = {
				fail_text = HAS_ABSOLUTE_ETHIC_PROJECT
				NOR = {
					has_special_project = ABSOLUTE_ETHIC_PROJECT
					has_special_project = ABSOLUTE_ETHIC_PROJECT_SPIRITUALIST
				}
			}
			if = {
				limit = { exists = ideology }
				OR = {
					custom_tooltip = {
						text = must_be_ideology_custodian
						ideology = { is_ideology_custodian = prev }
					}
					has_ascension_perk = ap_force_of_will
					AND = {
						custom_tooltip = {
							text = civic_tooltip_fan_authoritarian
							is_fanatic_authoritarian = yes
						}
						is_galactic_emperor = yes
					}
				}
				ideological_strength_percent >= 0.75
			}
			else = {
				custom_tooltip = {
					fail_text = HAS_ADOPTED_IDEOLOGY
					always = no
				}
			}
		}
	}
	else = {
		owner = {
			custom_tooltip = {
				success_text = civic_tooltip_not_fan_$ETHIC$
				fail_text = civic_tooltip_not_fan_$ETHIC$
				is_fanatic_$ETHIC$ = no
			}
		}
	}

	if = {
		limit = {
			exists = ideology
			owner = { has_ascension_perk = ap_force_of_will }
		}
		custom_tooltip = {
			fail_text = cannot_embrace_foreign_faction_while_has_ap_force_of_will
			owner = { exists = ideology }
			ideology = { is_same_value = prev.owner.ideology }
		}
	}
}

# Duplicate of can_embrace_faction without tooltips
can_embrace_faction_simplified = {
	support > 0.20
	owner = {
		OR = {
			is_$ETHIC$ = no
			is_crowdsourcing_empire = no
		}
		NOT = { has_modifier = embraced_faction_timer }
		NOT = { has_country_flag = political_reorientation }
		if = {
			limit = {
				prev = { is_faction_fanatic = yes }
				is_fanatic_$ETHIC$ = yes
			}
			NOT = { has_ethic = ethic_absolute_$ETHIC$ }
			NOR = {
				has_special_project = ABSOLUTE_ETHIC_PROJECT
				has_special_project = ABSOLUTE_ETHIC_PROJECT_SPIRITUALIST
			}
			exists = ideology
			ideological_strength_percent >= 0.75
		}
		else = { is_fanatic_$ETHIC$ = no }
	}

	if = {
		limit = {
			exists = ideology
			owner = { has_ascension_perk = ap_force_of_will }
		}
		owner = { exists = ideology }
		ideology? = { is_same_value = prev.owner.ideology }
	}
}

# Overlord, subject and federation ties rule out an Ideological Rivalry.
# This = declaring Country, $TARGET$ = the other Country
can_be_ideological_rival = {
	# Can't be rivals with your overlord or their other subjects.
	NAND = {
		is_subject = yes
		$TARGET$? = {
			OR = {
				is_fallen_empire = yes
				is_overlord_to = prev
				has_overlord = prev.overlord
			}
		}
	}
	# Can't be rivals with your subjects.
	NOT = {
		$TARGET$? = {
			is_subject = yes
			has_overlord = prev
		}
	}
	NOT = {
		$TARGET$? = { is_in_federation_with = prev }
	}
}
