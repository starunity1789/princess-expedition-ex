# 皇女的远征军EX 3.0：AI 自动星球的判断条件

# 国家作用域：只有自动星球的国家——常规 AI 帝国（含保底列强），游牧与荒野帝国除外
atb_ai_auto_country = {
	is_ai = yes
	is_country_type = default
	is_nomadic = no
	is_wilderness_empire = no
}

# 国家作用域：保底列强（开局就挂着 atbgp_power_base）
atb_ai_auto_great_power = {
	has_modifier = atbgp_power_base
}

# 星球作用域：已经是自动星球
atb_is_ai_auto_planet = {
	has_carrier_flag = atb_ai_auto
}

# 星球作用域：可以转成自动星球的殖民地（殖民已完成、主人适用、还没转过）
atb_ai_auto_convertible = {
	is_colony = yes
	exists = owner
	owner = { atb_ai_auto_country = yes }
	NOT = { has_carrier_flag = atb_ai_auto }
	pop_amount >= 100
	atb_ai_auto_supported_planet = yes
}

# 星球作用域：自动星球能放建筑的区划集（3.4）。宇宙创生世界、游牧等特殊区划集没有对应的城市区划，
# 6 座自动建筑放不进去，这类星球不转（已经转了的每年自检时放回普通星球）
atb_ai_auto_supported_planet = {
	OR = {
		uses_district_set = standard
		uses_district_set = habitat
		uses_district_set = ring_world
		uses_district_set = city_world
		uses_district_set = hive_world
		uses_district_set = machine_world
		uses_district_set = shattered_ring_world
		uses_district_set = volcanic_world
	}
}
