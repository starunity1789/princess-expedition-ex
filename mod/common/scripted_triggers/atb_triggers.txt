# 皇女的远征军：脚本触发器

# 国家作用域：AI 设计无畏舰时，没有可用的 T 槽武器就不选 T 槽舰首
# 湮灭光束要稀有水晶，远古湮灭眩光要远古学飞升和文物收入
atb_has_no_valid_t_weapon = {
	NOR = {
		AND = {
			has_technology = tech_titans
			has_monthly_income = {
				resource = rare_crystals
				value >= 2
			}
		}
		AND = {
			has_technology = tech_archaeo_titan_beam
			has_ascension_perk = ap_archaeoengineers
			has_monthly_income = {
				resource = minor_artifacts
				value >= 2
			}
		}
	}
}

# 预设帝国「苍神缘起帝国」可选的条件：与起源 origin_atb_princess_expedition 的 playable 相同
atb_preset_playable = {
	has_machine_age_dlc = yes
	has_ancrel = yes
}

# 国家作用域：可以通过决议探索的先驱者科技还没有全部得到
atb_has_unclaimed_precursor_secret = {
	NAND = {
		has_technology = tech_secrets_cybrex
		has_technology = tech_secrets_league
		has_technology = tech_secrets_irassian
		has_technology = tech_secrets_vultaum
		has_technology = tech_secrets_yuht
		has_technology = tech_secrets_baol
		has_technology = tech_secrets_zroni
	}
}

# 3.15：远征军融合「黄金世代」起源的学院与培养领袖部分（意志之力 DLC）。
# 远征军的起源是合成繁衍，所以原版黄金世代相关文件里的 has_origin = origin_golden_generation 都换成这个触发器
# （见 tools/gen_vanilla_patches.py）；国家标记 atb_golden_generation 在 atb_swap_origin 里设置，学院失去时原版事件会去掉它
# 国家作用域
atb_has_golden_generation = {
	OR = {
		has_origin = origin_golden_generation
		has_country_flag = atb_golden_generation
	}
}

# 3.24：普通 AI（含燃烧白鹿军团、远星同盟组织）一律走「开辟己道」，不与任何迷雾赞助者立约。

	NOR = {

	}
